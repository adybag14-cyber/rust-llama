//! Lossless GGUF wire strings with explicit native C-string lookup views.
//! The checked text facade consumes this same parser and retained reader.
use crate::{
    DType, Endian, Error, Gguf, HeaderReader, HeaderUsage, Limits, Parser, RawValue as Value,
    Result, TensorInfo, TensorReadInfo, Value as TextValue, align, dimension_elements, invalid,
    read_tensor_f32_info, read_tensor_payload_info,
};
use std::{
    collections::{BTreeMap, BTreeSet},
    fs::File,
    io::{Read, Seek, SeekFrom},
    path::Path,
};

pub(crate) fn c_string(bytes: &[u8]) -> &[u8] {
    &bytes[..bytes
        .iter()
        .position(|&byte| byte == 0)
        .unwrap_or(bytes.len())]
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct RawTensorInfo {
    /// The complete serialized name, used for header duplicate validation.
    pub name: Vec<u8>,
    pub dimensions: Vec<u64>,
    pub dtype: DType,
    pub offset: u64,
    pub byte_len: u64,
}
impl RawTensorInfo {
    /// Frozen GGML/getter name after C-string truncation.
    pub fn native_name(&self) -> &[u8] {
        c_string(&self.name)
    }
    pub fn elements(&self) -> Result<usize> {
        dimension_elements(&self.dimensions)
    }
    fn read_info(&self) -> TensorReadInfo<'_> {
        TensorReadInfo {
            name: String::from_utf8_lossy(&self.name),
            dimensions: &self.dimensions,
            dtype: self.dtype,
            offset: self.offset,
            byte_len: self.byte_len,
        }
    }
}

/// A retained GGUF reader whose keys, string values and tensor names are bytes.
/// Complete serialized names govern duplicate validation; native lookups use
/// C-string prefixes in original descriptor order. No lossy text conversion is
/// used to index or write the data.
pub struct RawGguf<R: Read + Seek> {
    pub(crate) reader: R,
    pub version: u32,
    pub endian: Endian,
    pub alignment: u32,
    pub metadata: BTreeMap<Vec<u8>, Value>,
    metadata_order: Vec<Vec<u8>>,
    pub tensors: Vec<RawTensorInfo>,
    pub data_offset: u64,
    pub file_len: u64,
    limits: Limits,
    header_usage: HeaderUsage,
}
impl RawGguf<File> {
    pub fn open(path: impl AsRef<Path>) -> Result<Self> {
        Self::read(File::open(path)?, Limits::default())
    }
    pub fn open_with_limits(path: impl AsRef<Path>, limits: Limits) -> Result<Self> {
        Self::read(File::open(path)?, limits)
    }
}
impl<R: Read + Seek> RawGguf<R> {
    /// Parse the header and validate every stored tensor against the file length.
    pub fn read(reader: R, limits: Limits) -> Result<Self> {
        Self::read_with_payload_validation(reader, limits, true)
    }
    /// Parse only the header, matching the frozen no_alloc GGUF parser.
    /// Later payload reads still perform checked file-bound validation.
    pub fn read_header(reader: R, limits: Limits) -> Result<Self> {
        Self::read_with_payload_validation(reader, limits, false)
    }
    pub(crate) fn read_with_payload_validation(
        mut reader: R,
        limits: Limits,
        validate_payload: bool,
    ) -> Result<Self> {
        let file_len = reader.seek(SeekFrom::End(0))?;
        reader.seek(SeekFrom::Start(0))?;
        let mut magic = [0; 4];
        reader.read_exact(&mut magic)?;
        if &magic != b"GGUF" {
            return Err(invalid("magic is not GGUF"));
        }
        let mut version_bytes = [0; 4];
        reader.read_exact(&mut version_bytes)?;
        let (endian, version) = if matches!(u32::from_le_bytes(version_bytes), 2 | 3) {
            (Endian::Little, u32::from_le_bytes(version_bytes))
        } else if matches!(u32::from_be_bytes(version_bytes), 2 | 3) {
            (Endian::Big, u32::from_be_bytes(version_bytes))
        } else {
            return Err(Error::Unsupported(format!(
                "version bytes {version_bytes:?}; supported versions are 2 and 3"
            )));
        };
        // Keep the public owner R unchanged. Only header parsing borrows a
        // bounded, fallibly allocated buffer; payload reads use the same R.
        let mut header_reader = HeaderReader::new(
            &mut reader,
            8,
            file_len.saturating_sub(8),
            limits.max_metadata_bytes.saturating_sub(8),
        )?;
        let parsed_header: Result<_> = (|| {
            let mut p = Parser {
                reader: &mut header_reader,
                endian,
                position: 8,
                file_len,
                limits: &limits,
                total_values: 0,
            };
            let nt = p.count(limits.max_tensors, "tensor count")?;
            let nk = p.count(limits.max_metadata_entries, "metadata count")?;
            p.minimum_encoded(nk, 14, "metadata entries")?;
            let mut metadata_order = Vec::new();
            metadata_order
                .try_reserve_exact(nk)
                .map_err(|_| Error::Limit("metadata order allocation".into()))?;
            let mut metadata = BTreeMap::new();
            for _ in 0..nk {
                let key = p.string()?;
                if key.is_empty() {
                    return Err(invalid("empty metadata key"));
                }
                let ty = p.u32()?;
                let value = p.value(ty, false)?;
                let mut ordered_key = Vec::new();
                ordered_key
                    .try_reserve_exact(key.len())
                    .map_err(|_| Error::Limit("metadata order key allocation".into()))?;
                ordered_key.extend_from_slice(&key);
                metadata_order.push(ordered_key);
                if metadata.contains_key(&key) {
                    return Err(invalid(format!("duplicate metadata key {key:?}")));
                }
                metadata.insert(key, value);
            }
            let alignment_key = metadata_order
                .iter()
                .find(|key| c_string(key) == b"general.alignment");
            let alignment = match alignment_key.and_then(|key| metadata.get(key)) {
                None => 32,
                Some(Value::U32(v)) => *v,
                _ => return Err(invalid("general.alignment must be UINT32")),
            };
            if !alignment.is_power_of_two() || alignment > 1024 * 1024 {
                return Err(invalid(
                    "alignment must be a power of two within 1..=1048576",
                ));
            }
            // Even a zero-dimensional tensor with an empty name needs 24
            // encoded descriptor bytes. Check both backing file and header budget
            // before reserving descriptor storage for an attacker-controlled count.
            p.minimum_encoded(nt, 24, "tensor descriptors")?;
            let mut tensors = Vec::new();
            tensors
                .try_reserve_exact(nt)
                .map_err(|_| Error::Limit("tensor descriptors allocation".into()))?;
            let mut names = BTreeSet::new();
            let mut expected_offset = 0u64;
            for _ in 0..nt {
                let name = p.string()?;
                if name.len() >= 64 || !names.insert(name.clone()) {
                    return Err(invalid("tensor name is too long or is duplicated"));
                }
                let nd = p.u32()?;
                if nd > 4 {
                    return Err(invalid(format!("tensor {name:?} has {nd} dimensions")));
                }
                let mut dimensions = Vec::new();
                dimensions
                    .try_reserve_exact(nd as usize)
                    .map_err(|_| Error::Limit("tensor dimensions allocation".into()))?;
                for _ in 0..nd {
                    let n = p.u64()?;
                    if n > i64::MAX as u64 {
                        return Err(invalid("invalid tensor dimension"));
                    }
                    dimensions.push(n)
                }
                let id = p.u32()?;
                let dtype = DType::from_id(id).map_err(|e| invalid(e.to_string()))?;
                dimension_elements(&dimensions)?;
                let row = usize::try_from(dimensions.first().copied().unwrap_or(1))
                    .map_err(|_| invalid("row size exceeds usize"))?;
                let row_bytes = dtype.row_bytes(row).map_err(|e| invalid(e.to_string()))? as u64;
                let byte_len = dimensions.iter().skip(1).try_fold(row_bytes, |a, &b| {
                    a.checked_mul(b)
                        .ok_or_else(|| invalid("tensor byte size overflow"))
                })?;
                let offset = p.u64()?;
                // Frozen gguf.cpp requires descriptor-order contiguous padded payloads,
                // not merely aligned, non-overlapping regions.
                if offset != expected_offset {
                    return Err(invalid(format!(
                        "tensor {name:?} offset {offset} differs from expected {expected_offset} (gap, overlap, or reordering)"
                    )));
                }
                expected_offset = expected_offset
                    .checked_add(align(byte_len, alignment as u64)?)
                    .ok_or_else(|| invalid("padded tensor data size overflow"))?;
                tensors.push(RawTensorInfo {
                    name,
                    dimensions,
                    dtype,
                    offset,
                    byte_len,
                });
            }
            // The frozen reader does not align the reported data offset for an
            // empty tensor directory, even when the writer included padding.
            let data_offset = if nt == 0 {
                p.position
            } else {
                align(p.position, alignment as u64)?
            };
            if data_offset > limits.max_metadata_bytes {
                return Err(Error::Limit(
                    "metadata byte budget including alignment".into(),
                ));
            }
            let header_usage = HeaderUsage {
                bytes: data_offset,
                metadata_entries: nk,
                metadata_values: p.total_values,
                tensors: nt,
            };
            for t in &tensors {
                let start = data_offset
                    .checked_add(t.offset)
                    .ok_or_else(|| invalid("tensor absolute offset overflow"))?;
                let end = start
                    .checked_add(t.byte_len)
                    .ok_or_else(|| invalid("tensor end overflow"))?;
                if validate_payload && end > file_len {
                    return Err(invalid(format!("tensor {:?} exceeds file length", t.name)));
                }
            }
            Ok((
                metadata,
                metadata_order,
                tensors,
                alignment,
                data_offset,
                header_usage,
            ))
        })();
        // Rewind bounded read-ahead on success and failure. Preserve the
        // original parse error if synchronization also fails.
        let synchronized = header_reader.finish();
        let (metadata, metadata_order, tensors, alignment, data_offset, header_usage) =
            parsed_header?;
        synchronized?;
        Ok(Self {
            reader,
            version,
            endian,
            alignment,
            metadata,
            metadata_order,
            tensors,
            data_offset,
            file_len,
            limits,
            header_usage,
        })
    }
    pub fn metadata_order(&self) -> impl Iterator<Item = &[u8]> {
        self.metadata_order
            .iter()
            .filter(|key| self.metadata.contains_key(*key))
            .map(Vec::as_slice)
    }
    /// First original-order key matching the native C-string prefix.
    pub fn metadata_native(&self, name: &[u8]) -> Option<&Value> {
        let wanted = c_string(name);
        self.metadata_order()
            .find(|key| c_string(key) == wanted)
            .and_then(|key| self.metadata.get(key))
    }
    pub fn header_usage(&self) -> HeaderUsage {
        self.header_usage
    }
    pub fn find_tensor(&self, name: &[u8]) -> Option<usize> {
        let wanted = c_string(name);
        self.tensors
            .iter()
            .rposition(|tensor| tensor.native_name() == wanted)
    }
    pub fn tensor(&self, name: &[u8]) -> Option<&RawTensorInfo> {
        self.find_tensor(name)
            .and_then(|index| self.tensors.get(index))
    }
    pub fn read_tensor(&mut self, name: &[u8]) -> Result<Vec<u8>> {
        let index = self
            .find_tensor(name)
            .ok_or_else(|| invalid(format!("missing byte-named tensor {name:?}")))?;
        self.read_tensor_by_index(index)
    }
    pub fn read_tensor_by_index(&mut self, index: usize) -> Result<Vec<u8>> {
        let tensor = self
            .tensors
            .get(index)
            .ok_or_else(|| invalid("tensor descriptor index is outside the file"))?;
        read_tensor_payload_info(
            &mut self.reader,
            self.data_offset,
            self.file_len,
            &self.limits,
            &tensor.read_info(),
        )
    }
    pub fn read_f32(&mut self, name: &[u8]) -> Result<Vec<f32>> {
        let index = self
            .find_tensor(name)
            .ok_or_else(|| invalid(format!("missing byte-named tensor {name:?}")))?;
        let tensor = &self.tensors[index];
        read_tensor_f32_info(
            &mut self.reader,
            self.data_offset,
            self.file_len,
            self.endian,
            &self.limits,
            &tensor.read_info(),
        )
    }
    /// Serialize this header using the frozen C reader/writer representation.
    /// Raw keys and values retain their complete lengths; tensor names use
    /// C-string prefixes and dimensions use GGML rank. This can reproduce the
    /// original's invalid serialized output for accepted prefix collisions.
    /// Newly inserted public-map keys have no original position; use
    /// write_raw_header with an explicit order for constructed metadata.
    pub fn write_native_header<W: std::io::Write>(
        &self,
        out: &mut W,
    ) -> Result<crate::WriteLayout> {
        if self.metadata.len() > self.limits.max_metadata_entries
            || self.tensors.len() > self.limits.max_tensors
        {
            return Err(Error::Limit("native header index budget".into()));
        }
        let mut metadata = Vec::new();
        metadata
            .try_reserve_exact(self.metadata_order.len())
            .map_err(|_| Error::Limit("native header metadata index allocation".into()))?;
        metadata.extend(self.metadata_order().map(|key| (key, &self.metadata[key])));
        let minimum = (self.tensors.len() as u64)
            .checked_mul(24)
            .ok_or_else(|| Error::Limit("native header descriptor minimum overflow".into()))?;
        if minimum > self.limits.max_metadata_bytes.saturating_sub(24) {
            return Err(Error::Limit(
                "native header descriptor minimum exceeds budget".into(),
            ));
        }
        let mut tensors = Vec::new();
        tensors
            .try_reserve_exact(self.tensors.len())
            .map_err(|_| Error::Limit("native header tensor index allocation".into()))?;
        tensors.extend(
            self.tensors
                .iter()
                .map(|tensor| crate::RawTensorDescriptor {
                    name: tensor.name.as_slice(),
                    dimensions: &tensor.dimensions,
                    dtype: tensor.dtype,
                }),
        );
        crate::writer::native_raw_header(
            out,
            &metadata,
            &tensors,
            &self.limits,
            self.version,
            self.alignment,
        )
    }

    /// Consume the same owner into the existing UTF-8/non-NUL text facade.
    /// Failure releases the staged owner; no partially converted file escapes.
    pub fn into_text(self) -> Result<Gguf<R>> {
        let mut metadata = BTreeMap::new();
        for (key, value) in self.metadata {
            if key.contains(&0) {
                return Err(invalid("empty/NUL metadata key"));
            }
            let key = text(key)?;
            metadata.insert(key, text_value(value)?);
        }
        let mut metadata_order = Vec::new();
        metadata_order
            .try_reserve_exact(self.metadata_order.len())
            .map_err(|_| Error::Limit("text metadata order allocation".into()))?;
        for key in self.metadata_order {
            metadata_order.push(text(key)?);
        }
        let mut tensors = Vec::new();
        tensors
            .try_reserve_exact(self.tensors.len())
            .map_err(|_| Error::Limit("text tensor descriptor allocation".into()))?;
        for tensor in self.tensors {
            if tensor.name.contains(&0) {
                return Err(invalid("tensor name contains NUL"));
            }
            tensors.push(TensorInfo {
                name: text(tensor.name)?,
                dimensions: tensor.dimensions,
                dtype: tensor.dtype,
                offset: tensor.offset,
                byte_len: tensor.byte_len,
            });
        }
        Ok(Gguf {
            reader: self.reader,
            version: self.version,
            endian: self.endian,
            alignment: self.alignment,
            metadata,
            metadata_order,
            tensors,
            data_offset: self.data_offset,
            file_len: self.file_len,
            limits: self.limits,
            header_usage: self.header_usage,
        })
    }
}
fn text(bytes: Vec<u8>) -> Result<String> {
    String::from_utf8(bytes).map_err(|_| invalid("non-UTF-8 string"))
}
fn text_value(value: Value) -> Result<TextValue> {
    Ok(match value {
        Value::U8(v) => TextValue::U8(v),
        Value::I8(v) => TextValue::I8(v),
        Value::U16(v) => TextValue::U16(v),
        Value::I16(v) => TextValue::I16(v),
        Value::U32(v) => TextValue::U32(v),
        Value::I32(v) => TextValue::I32(v),
        Value::F32(v) => TextValue::F32(v),
        Value::Bool(v) => TextValue::Bool(v),
        Value::String(v) => TextValue::String(text(v)?),
        Value::Array {
            element_type,
            values,
        } => {
            let mut result = Vec::new();
            result
                .try_reserve_exact(values.len())
                .map_err(|_| Error::Limit("text metadata array allocation".into()))?;
            for item in values {
                result.push(text_value(item)?);
            }
            TextValue::Array {
                element_type,
                values: result,
            }
        }
        Value::U64(v) => TextValue::U64(v),
        Value::I64(v) => TextValue::I64(v),
        Value::F64(v) => TextValue::F64(v),
    })
}

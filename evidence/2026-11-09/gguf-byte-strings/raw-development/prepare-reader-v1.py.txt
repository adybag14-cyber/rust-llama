from pathlib import Path
import json,hashlib,subprocess,datetime
root=Path(r'C:\Users\adyba\rust-llama-reference\gguf-byte-strings');repo=root/'rust-draft-v1'
out=root/'prepare-reader-v1';out.mkdir(exist_ok=False)
sha=lambda b:hashlib.sha256(b).hexdigest()
seed=json.loads((root/'seed-v1.json').read_text(encoding='utf-8'))
payload=json.loads((root/'source-payload-v1.json').read_text(encoding='utf-8'))
report={'status':'running','started_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'seed_sha256':sha((root/'seed-v1.json').read_bytes()),'payload_sha256':sha((root/'source-payload-v1.json').read_bytes()),'source_changes':[]}
def save():(out/'status.json').write_bytes((json.dumps(report,indent=2)+'\n').encode('utf-8'))
save()
try:
 for cohort in ['original-reader-v1','original-booleans-v1']:
  sealed=json.loads((root/cohort/'sealed-review.json').read_text(encoding='utf-8'));assert sealed['status']=='succeeded'
  for record in sealed['files']:
   p=root/record['path'];assert p.stat().st_size==record['bytes'] and sha(p.read_bytes())==record['sha256']
 assert sha((repo/'crates/gguf/src/lib.rs').read_bytes())==payload['preimage_sha256']==seed['source_preimages']['crates/gguf/src/lib.rs']['sha256']
 assert not (repo/'crates/gguf/src/raw.rs').exists()
 for entry in payload['files']:
  p=repo/entry['path'];before=p.read_bytes() if p.exists() else None
  if before is not None:
   old=out/'preimages'/entry['path'];old.parent.mkdir(parents=True,exist_ok=True);old.write_bytes(before)
  data=entry['text'].encode('utf-8');p.write_bytes(data)
  report['source_changes'].append({'path':entry['path'],'before':None if before is None else {'bytes':len(before),'sha256':sha(before)},'applied':{'bytes':len(data),'sha256':sha(data)}});save()
 p=subprocess.run(['cargo','+1.99.1','fmt','--all'],cwd=repo,capture_output=True,timeout=120)
 (out/'fmt.stdout.log').write_bytes(p.stdout);(out/'fmt.stderr.log').write_bytes(p.stderr)
 assert p.returncode==0,p.stderr.decode('utf-8','replace')
 for record in report['source_changes']:
  p=repo/record['path'];record['formatted']={'bytes':p.stat().st_size,'sha256':sha(p.read_bytes())}
 report['status']='succeeded'
except BaseException as e:report['status']='failed';report['error']=repr(e);raise
finally:report['finished_utc']=datetime.datetime.now(datetime.timezone.utc).isoformat();save()
print(json.dumps(report,indent=2))

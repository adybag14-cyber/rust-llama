from pathlib import Path
import json,hashlib,subprocess,os,datetime
base=Path(r'C:\Users\adyba\rust-llama-reference\gguf-byte-strings')
repo=base/'rust-draft-v1';out=base/'prepare-writer-v1';out.mkdir(exist_ok=False)
data=json.loads((base/'raw-writer-payload-v1.json').read_text(encoding='utf-8'))
sha=lambda b:hashlib.sha256(b).hexdigest()
paths={name:repo/'crates/gguf/src'/(name+'.rs') for name in ['writer','raw','lib']}
for name,p in paths.items():
 assert sha(p.read_bytes())==data[name+'_preimage_sha256'],name
 (out/(name+'.preimage.rs.txt')).write_bytes(p.read_bytes())
lib=paths['lib'].read_text(encoding='utf-8')
old='pub use writer::{TensorDescriptor, WriteLayout, write_header, write_ordered};'
assert lib.count(old)==1
lib=lib.replace(old,'pub use writer::{RawTensorDescriptor, TensorDescriptor, WriteLayout, write_header, write_ordered, write_raw_header};')
raw=paths['raw'].read_text(encoding='utf-8')
marker='    /// Consume the same owner into the existing UTF-8/non-NUL text facade.'
assert raw.count(marker)==1
raw=raw.replace(marker,data['native_header_method']+'\n'+marker)
paths['writer'].write_bytes(data['writer'].encode('utf-8'));paths['lib'].write_bytes(lib.encode('utf-8'));paths['raw'].write_bytes(raw.encode('utf-8'))
p=subprocess.run(['cargo','+1.99.1','fmt','--all'],cwd=repo,stdout=subprocess.PIPE,stderr=subprocess.STDOUT)
(out/'fmt.log').write_bytes(p.stdout);assert p.returncode==0,p.stdout
report={'status':'succeeded','created_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),
 'files':[{'path':p.relative_to(repo).as_posix(),'before_sha256':data[name+'_preimage_sha256'],'after_sha256':sha(p.read_bytes()),'after_bytes':p.stat().st_size} for name,p in paths.items()]}
(out/'status.json').write_bytes((json.dumps(report,indent=2)+'\n').encode('utf-8'));print(json.dumps(report,indent=2))

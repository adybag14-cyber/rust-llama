+/// Header-only read-ahead. The owned file/reader remains in Gguf<R>.
+/// Unlike an unbounded convenience buffer, this allocation is fallible and
+/// each refill is capped by the remaining physical file and header allowance.
+struct HeaderReader<'a, R: Read + Seek> {
+    reader: &'a mut R,
+    buffer: Vec<u8>,
+    begin: usize,
+    end: usize,
+    remaining: u64,
+    position: u64,
+}
+impl<'a, R: Read + Seek> HeaderReader<'a, R> {
+    fn new(
+        reader: &'a mut R,
+        position: u64,
+        file_remaining: u64,
+        header_remaining: u64,
+    ) -> Result<Self> {
+        let remaining = file_remaining.min(header_remaining);
+        let capacity = remaining.min(64 * 1024) as usize;
+        let mut buffer = Vec::new();
+        buffer
+            .try_reserve_exact(capacity)
+            .map_err(|_| Error::Limit("GGUF header read buffer allocation".into()))?;
+        buffer.resize(capacity, 0);
+        Ok(Self {
+            reader,
+            buffer,
+            begin: 0,
+            end: 0,
+            remaining,
+            position,
+        })
+    }
+    fn finish(self) -> Result<()> {
+        self.reader.seek(SeekFrom::Start(self.position))?;
+        Ok(())
+    }
+}
+impl<R: Read + Seek> Read for HeaderReader<'_, R> {
+    fn read(&mut self, output: &mut [u8]) -> io::Result<usize> {
+        if output.is_empty() {
+            return Ok(0);
+        }
+        if self.begin == self.end {
+            if self.remaining == 0 {
+                return Ok(0);
+            }
+            let request = self.remaining.min(self.buffer.len() as u64) as usize;
+            let received = self.reader.read(&mut self.buffer[..request])?;
+            if received > request {
+                return Err(io::Error::new(
+                    io::ErrorKind::InvalidData,
+                    "reader returned more bytes than requested",
+                ));
+            }
+            self.remaining -= received as u64;
+            self.begin = 0;
+            self.end = received;
+        }
+        let count = output.len().min(self.end - self.begin);
+        output[..count].copy_from_slice(&self.buffer[self.begin..self.begin + count]);
+        self.begin += count;
+        self.position += count as u64;
+        Ok(count)
+    }

+        let resolve = |index: usize| paths.get(index).map(AsRef::as_ref);
+        Self::load(first.as_ref(), Some((paths.len(), &resolve)), limits)
+    }
+
+    fn load<'a>(
+        first_path: &Path,
+        explicit: Option<ExplicitSplits<'a, '_>>,
+        limits: ShardLimits,
+    ) -> Result<Self> {
+        if limits.max_shards == 0 {
+            return Err(Error::Limit("maximum GGUF shard count is zero".into()));
+        }
+        let mut first = Gguf::open_with_limits(first_path, limits.gguf.clone())?;
+        let declared_count = u16_key(&first.metadata, "split.count", Some(0))?;
+        let count = usize::from(declared_count).max(1);
+        if count > limits.max_shards {
+            return Err(Error::Limit(format!(
+                "model declares {count} GGUF shards, exceeding {}",
+                limits.max_shards
+            )));
+        }
+        let prefix = if declared_count > 1 {
+            if u16_key(&first.metadata, "split.no", None)? != 0 {
+                return Err(invalid(
+                    "model must be loaded with the first split (split.no=0)",
+                ));
+            }
+            if let Some((given, _)) = explicit {
+                // Deliberately no UINT16 narrowing: the source's cast can wrap
+                // for oversized lists; that is not a safe input convention.
+                if given != count {
+                    return Err(invalid(format!(
+                        "explicit split count {given} differs from declared {count}"
+                    )));
+                }
+                None
+            } else {
+                Some(split_prefix(first_path, 0, declared_count)?)
+            }
+        } else {
+            None
+        };
+
+        let mut set = Self {
+            metadata: std::mem::take(&mut first.metadata),
+            shards: Vec::new(),
+            tensors: Vec::new(),
+            tensor_shards: Vec::new(),
+            by_name: Vec::new(),
+            usage: HeaderUsage::default(),
+            file_bytes: 0,
+            tensor_bytes: 0,
+            tensor_elements: 0,
+        };
+        set.shards
+            .try_reserve_exact(count)
+            .map_err(|_| Error::Limit("GGUF shard owners allocation".into()))?;
+        set.append(first_path, first)?;
+        for index in 1..count {
+            let path = match explicit {
+                Some((_, resolve)) => copy_path(
+                    resolve(index).ok_or_else(|| invalid("explicit split path is missing"))?,
+                )?,
+                None => split_path(
+                    prefix
+                        .as_deref()
+                        .ok_or_else(|| invalid("missing split filename prefix"))?,
+                    index,
+                    declared_count,
+                )?,
+            };
+            let remaining = remaining_limits(&limits.gguf, set.usage)?;
+            let file = Gguf::open_with_limits(&path, remaining)?;
+            // This is the ONLY subsidiary model metadata field consulted by
+            // the original loader. In particular split.count, total, arch,
+            // tokenizer and other primary metadata must not be merged.
+            if usize::from(u16_key(&file.metadata, "split.no", None)?) != index {
+                return Err(invalid(format!("split.no differs from expected {index}")));
+            }
+            set.append(&path, file)?;
+        }
+        set.by_name
+            .try_reserve_exact(set.tensors.len())
+            .map_err(|_| Error::Limit("unified tensor name index allocation".into()))?;
+        set.by_name.extend(0..set.tensors.len());
+        set.by_name
+            .sort_unstable_by(|&left, &right| set.tensors[left].name.cmp(&set.tensors[right].name));
+        for pair in set.by_name.windows(2) {
+            if set.tensors[pair[0]].name == set.tensors[pair[1]].name {
+                return Err(invalid(format!(
+                    "tensor {} is duplicated across GGUF splits",
+                    set.tensors[pair[0]].name
+                )));
+            }
+        }
+        if declared_count > 1 {
+            let expected = match set.metadata.get("split.tensors.count") {
+                Some(Value::I32(value)) if *value >= 0 => *value as usize,
+                Some(Value::I32(_)) => return Err(invalid("split.tensors.count is negative")),
+                Some(_) => return Err(invalid("split.tensors.count must be INT32")),
+                None => return Err(invalid("missing split.tensors.count")),
+            };
+            if expected != set.tensors.len() {
+                return Err(invalid(format!(
+                    "split.tensors.count={expected}, but {} tensors were found",
+                    set.tensors.len()
+                )));
+            }
+        }
+        Ok(set)
+    }
+
+    fn append(&mut self, path: &Path, mut file: Gguf<File>) -> Result<()> {
+        let shard_index = self.shards.len();
+        self.usage = add_usage(self.usage, file.header_usage)?;
+        self.file_bytes = self
+            .file_bytes
+            .checked_add(file.file_len)
+            .ok_or_else(|| invalid("aggregate GGUF file size overflow"))?;
+        self.tensors
+            .try_reserve(file.tensors.len())
+            .map_err(|_| Error::Limit("unified tensor descriptors allocation".into()))?;
+        self.tensor_shards
+            .try_reserve(file.tensors.len())
+            .map_err(|_| Error::Limit("unified tensor origins allocation".into()))?;
+        for tensor in &file.tensors {
+            self.tensor_bytes = self
+                .tensor_bytes
+                .checked_add(tensor.byte_len)
+                .ok_or_else(|| invalid("aggregate tensor byte count overflow"))?;
+            self.tensor_elements = self
+                .tensor_elements
+                .checked_add(tensor.elements()? as u64)
+                .ok_or_else(|| invalid("aggregate tensor element count overflow"))?;
+        }
+        let info = ShardInfo {
+            path: copy_path(path)?,
+            version: file.version,
+            endian: file.endian,
+            alignment: file.alignment,
+            data_offset: file.data_offset,
+            file_len: file.file_len,
+            tensor_count: file.tensors.len(),
+            header_usage: file.header_usage,
+        };
+        self.tensor_shards
+            .extend(std::iter::repeat_n(shard_index, file.tensors.len()));
+        self.tensors.append(&mut file.tensors);
+        // Secondary headers have already been validated and charged. Retain
+        // only primary metadata and file-specific layout, as the source does.
+        file.metadata.clear();
+        self.shards.push(Shard { info, file });
+        Ok(())
+    }

+fn add_usage(a: HeaderUsage, b: HeaderUsage) -> Result<HeaderUsage> {
+    let overflow = || Error::Limit("cumulative GGUF header accounting overflow".into());
+    Ok(HeaderUsage {
+        bytes: a.bytes.checked_add(b.bytes).ok_or_else(overflow)?,
+        metadata_entries: a
+            .metadata_entries
+            .checked_add(b.metadata_entries)
+            .ok_or_else(overflow)?,
+        metadata_values: a
+            .metadata_values
+            .checked_add(b.metadata_values)
+            .ok_or_else(overflow)?,
+        tensors: a.tensors.checked_add(b.tensors).ok_or_else(overflow)?,
+    })
+}
+fn remaining_limits(limits: &Limits, used: HeaderUsage) -> Result<Limits> {
+    let exhausted = || Error::Limit("cumulative GGUF header budget".into());
+    Ok(Limits {
+        max_metadata_bytes: limits
+            .max_metadata_bytes
+            .checked_sub(used.bytes)
+            .ok_or_else(exhausted)?,
+        max_total_values: limits
+            .max_total_values
+            .checked_sub(used.metadata_values)
+            .ok_or_else(exhausted)?,
+        max_tensors: limits
+            .max_tensors
+            .checked_sub(used.tensors)
+            .ok_or_else(exhausted)?,
+        max_metadata_entries: limits
+            .max_metadata_entries
+            .checked_sub(used.metadata_entries)
+            .ok_or_else(exhausted)?,
+        ..limits.clone()
+    })

+fn copy_path(path: &Path) -> Result<PathBuf> {
+    let mut result = OsString::new();
+    result
+        .try_reserve_exact(path.as_os_str().len())
+        .map_err(|_| Error::Limit("GGUF path allocation".into()))?;
+    result.push(path.as_os_str());
+    Ok(result.into())
+}
+fn split_path(prefix: &Path, index: usize, count: u16) -> Result<PathBuf> {
+    let suffix = format!("-{:05}-of-{count:05}.gguf", index + 1);
+    let mut path = copy_path(prefix)?.into_os_string();
+    path.try_reserve_exact(suffix.len())
+        .map_err(|_| Error::Limit("GGUF split path allocation".into()))?;
+    path.push(suffix);
+    Ok(path.into())
+}
+fn split_prefix(path: &Path, index: usize, count: u16) -> Result<PathBuf> {
+    let suffix = format!("-{:05}-of-{count:05}.gguf", index + 1);
+    strip_suffix(path, &suffix)
+        .ok_or_else(|| invalid(format!("invalid split filename: {}", path.display())))?
+}

+#[cfg(unix)]
+fn strip_suffix(path: &Path, suffix: &str) -> Option<Result<PathBuf>> {
+    use std::os::unix::ffi::{OsStrExt, OsStringExt};
+    let original = path.as_os_str().as_bytes();
+    let prefix = original.strip_suffix(suffix.as_bytes())?;
+    if prefix.is_empty() {
+        return None;
+    }
+    let mut bytes = Vec::new();
+    if bytes.try_reserve_exact(prefix.len()).is_err() {
+        return Some(Err(Error::Limit("GGUF split prefix allocation".into())));
+    }
+    bytes.extend_from_slice(prefix);
+    Some(Ok(OsString::from_vec(bytes).into()))
+}
+#[cfg(windows)]
+fn strip_suffix(path: &Path, suffix: &str) -> Option<Result<PathBuf>> {
+    use std::os::windows::ffi::{OsStrExt, OsStringExt};
+    let mut encoded = Vec::new();
+    if encoded.try_reserve_exact(path.as_os_str().len()).is_err() {
+        return Some(Err(Error::Limit("GGUF split prefix allocation".into())));
+    }
+    encoded.extend(path.as_os_str().encode_wide());
+    let n = encoded.len().checked_sub(suffix.len())?;
+    if n == 0
+        || !encoded[n..]
+            .iter()
+            .copied()
+            .eq(suffix.bytes().map(u16::from))
+    {
+        return None;
+    }
+    Some(Ok(OsString::from_wide(&encoded[..n]).into()))
+}
+#[cfg(not(any(unix, windows)))]
+fn strip_suffix(path: &Path, suffix: &str) -> Option<Result<PathBuf>> {
+    let prefix = path.to_str()?.strip_suffix(suffix)?;
+    if prefix.is_empty() {
+        None
+    } else {
+        Some(copy_path(Path::new(prefix)))
+    }

+#[derive(Default)]
+struct Metrics {
+    values: usize,
+    bitwise: usize,
+    maximum: f64,
+    largest_fraction: f64,
+    greedy: usize,
+}
+fn compare_values(actual: &[f32], expected: &[u8], label: &str, metrics: &mut Metrics) {
+    assert_eq!(expected.len(), actual.len() * 4, "{label}");
+    for (&a, e) in actual.iter().zip(expected.as_chunks::<4>().0) {
+        let e = f32::from_le_bytes(*e);
+        let difference = (f64::from(a) - f64::from(e)).abs();
+        let budget = 2e-4 + 2e-5 * f64::from(e).abs();
+        assert!(
+            a.is_finite() && e.is_finite() && difference <= budget,
+            "{label}: {a:.9e} != {e:.9e}; difference={difference:.9e} budget={budget:.9e}"
+        );
+        metrics.values += 1;
+        metrics.bitwise += usize::from(a.to_bits() == e.to_bits());
+        metrics.maximum = metrics.maximum.max(difference);
+        metrics.largest_fraction = metrics.largest_fraction.max(difference / budget);
+    }
+}
+fn ids(value: &Value) -> Vec<i32> {
+    array(value)
+        .iter()
+        .map(|v| i32::try_from(v.as_i64().unwrap()).unwrap())
+        .collect()
+}
+fn check_model(
+    model: &Model,
+    reference: &Value,
+    values: &[u8],
+    label: &str,
+    metrics: &mut Metrics,
+) {
+    assert_eq!(model.vocabulary_size(), number(field(reference, "n_vocab")));
+    let embedding = string(field(reference, "family")) == "embedding";
+    assert_eq!(model.supports_embeddings(), embedding);
+    assert_eq!(model.supports_generation(), !embedding);
+    for case in array(field(reference, "cases")) {
+        let tokens = ids(field(case, "tokens"));
+        let prompt = string(field(case, "prompt"));
+        assert_eq!(
+            model.tokenize(prompt, true).unwrap(),
+            tokens,
+            "{label} {prompt:?}"
+        );
+        if embedding {
+            assert_eq!(
+                model.embedding_dimensions(),
+                number(field(reference, "n_embd"))
+            );
+            let actual = model
+                .embed_tokens(&tokens, Pooling::None, &|| false)
+                .unwrap();
+            assert!(actual.pooled.is_none());
+            let expected = field(case, "embeddings");
+            assert_eq!(actual.tokens.len(), number(field(expected, "rows")));
+            let bytes = block(values, expected);
+            let width = number(field(expected, "columns"));
+            for (index, row) in actual.tokens.iter().enumerate() {
+                compare_values(
+                    row,
+                    &bytes[index * width * 4..(index + 1) * width * 4],
+                    label,
+                    metrics,
+                );
+            }
+            for (name, pooling) in [
+                ("mean", Pooling::Mean),
+                ("first", Pooling::First),
+                ("last", Pooling::Last),
+            ] {
+                let result = model.embed_tokens(&tokens, pooling, &|| false).unwrap();
+                compare_values(
+                    result.pooled.as_ref().unwrap(),
+                    block(values, field(field(case, "pooling"), name)),
+                    label,
+                    metrics,
+                );
+            }
+        } else {
+            let greedy = ids(field(case, "greedy_tokens"));
+            let mut sequence = tokens.clone();
+            sequence.extend_from_slice(&greedy[..greedy.len() - 1]);
+            let actual = model.logits_for_tokens(&sequence).unwrap();
+            let expected = field(case, "logits");
+            assert_eq!(actual.len(), number(field(expected, "rows")));
+            let bytes = block(values, expected);
+            let width = number(field(expected, "columns"));
+            for (index, row) in actual.iter().enumerate() {
+                compare_values(
+                    row,
+                    &bytes[index * width * 4..(index + 1) * width * 4],
+                    label,
+                    metrics,
+                );
+            }
+            for (step, &expected) in greedy.iter().enumerate() {
+                let row = &actual[tokens.len() - 1 + step];
+                let mut best = 0;
+                for i in 1..row.len() {
+                    if row[i] > row[best] {
+                        best = i;
+                    }
+                }
+                assert_eq!(best as i32, expected, "{label} greedy step {step}");
+                assert_eq!(
+                    model.tokenizer().token_piece(expected, false).unwrap(),
+                    decode_hex(string(&array(field(case, "greedy_pieces_hex"))[step]))
+                );
+                metrics.greedy += 1;
+            }
+        }
+    }
+}

+fn require_limit(result: llama_gguf::Result<GgufSet>) {
+    match result {
+        Err(llama_gguf::Error::Limit(_)) => {}
+        Err(error) => panic!("expected cumulative resource limit, got {error}"),
+        Ok(_) => panic!("expected cumulative resource limit"),
+    }
+}
+#[test]
+fn cumulative_limits_include_every_shard_and_reads_remain_lazy() {
+    let corpus = Corpus::new();
+    let case = corpus.case("three-explicit");
+    let paths = corpus.paths(case);
+    let original = GgufSet::open_splits(&paths).unwrap();
+    let usage = original.header_usage();
+    let exact = ShardLimits {
+        max_shards: paths.len(),
+        gguf: Limits {
+            max_metadata_bytes: usage.bytes,
+            max_metadata_entries: usage.metadata_entries,
+            max_total_values: usage.metadata_values,
+            max_tensors: usage.tensors,
+            ..Limits::default()
+        },
+    };
+    let loaded = GgufSet::open_splits_with_limits(&paths, exact.clone()).unwrap();
+    assert_eq!(loaded.header_usage(), usage);
+    for budget in 0..5 {
+        let mut limits = exact.clone();
+        match budget {
+            0 => limits.max_shards -= 1,
+            1 => limits.gguf.max_metadata_bytes -= 1,
+            2 => limits.gguf.max_metadata_entries -= 1,
+            3 => limits.gguf.max_total_values -= 1,
+            4 => limits.gguf.max_tensors -= 1,
+            _ => unreachable!(),
+        }
+        require_limit(GgufSet::open_splits_with_limits(&paths, limits));
+    }
+    let mut tiny_read = exact;
+    tiny_read.gguf.max_tensor_read_bytes = 1;
+    let mut lazy = GgufSet::open_splits_with_limits(&paths, tiny_read).unwrap();
+    assert!(matches!(
+        lazy.read_tensor("token_embd.weight"),
+        Err(llama_gguf::Error::Limit(_))
+    ));
+    assert!(matches!(
+        lazy.read_f32("token_embd.weight"),
+        Err(llama_gguf::Error::Limit(_))
+    ));
+    let model = Model::load_from_gguf(loaded, &LoadOptions::default()).unwrap();
+    assert_eq!(model.vocabulary_size(), 512);
+}

+fn rename_with_prefix(corpus: &Corpus, prefix: &std::ffi::OsStr) -> Vec<PathBuf> {
+    let source = corpus.paths(corpus.case("two-explicit"));
+    source
+        .iter()
+        .enumerate()
+        .map(|(index, input)| {
+            let mut name = prefix.to_owned();
+            name.push(format!("-{:05}-of-{:05}.gguf", index + 1, source.len()));
+            let path = corpus.root.join(name);
+            fs::copy(input, &path).unwrap();
+            path
+        })
+        .collect()
+}

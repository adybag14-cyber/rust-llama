+# Standard OS integration boundaries
+
+This crate contains the narrow Rust boundaries for native driver loading and
+system scalar math. It does not load llama/ggml or compile any project C/C++.
+
+## System scalar ERF
+
+The frozen CPU GELU_ERF uses the operating system's C-runtime
+`float erff(float)`. `math::erf_f32` exposes the same function through a safe
+`Option<f32>` API. The only unsafe operation is an audited call with one F32
+argument and one F32 result; no pointers, ownership, callbacks or allocation
+cross this boundary.
+
+| Build target | Linkage | Independently exercised runtime |
+| --- | --- | --- |
+| x86_64 Windows MSVC, dynamic CRT | Standard `ucrt` import library | `ucrtbase.dll` 10.0.26100.9711 |
+| x86_64 Linux GNU, dynamic CRT | Standard `-lm` system import | glibc libm 2.35 |
+| Other targets or static CRT | No ERF import from this module | Returns `None`; caller chooses its explicit fallback |
+
+Target selection identifies a linkage profile, not a guarantee for every future
+version of its runtime. `math::erf_provider` reports the selected provider.
+Tensor execution uses the existing independently bounded pure Rust
+mathematical ERF when this module declines a provider.
+
+The two original operating-system captures each contain **502,058 raw F32
+inputs and outputs**, including all half/bfloat encodings represented in F32,
+dense finite points, seeded raw bits, signed zero, infinities and NaN payloads.
+Both debug and release regression tests require exact output words against
+the matching host's independently captured reference. The fixtures and their
+provider/source hashes are under `tests/fixtures`. An additional original
+GGML graph corpus checks the enclosing GELU_ERF and GEGLU_ERF computations.
+
+The numerical environment is nearest-even rounding with gradual underflow.
+This API does not change the floating environment or promise floating exception
+status-flag parity. Cross-CRT equality, unmeasured runtime versions and other
+target profiles remain separate verification requirements.
+
+This addition introduces no third-party Cargo dependency, build script, C shim,
+vendor math library, or original-project binary into the Rust product.

+    if cfg!(all(
+        target_arch = "x86_64",
+        target_os = "linux",
+        target_env = "gnu"

+        MathPolicy::Sse2
+    } else {
+        MathPolicy::Scalar
+    }

+/// SiLU on one source row. Do not split a row into arbitrary scalar calls.

+/// Select the vector-prefix policy explicitly. Scalar tail exp uses the target
+/// platform's normal Rust/libm lowering, even for an explicitly selected policy.
+pub fn silu_f32_with_policy(values: &mut [f32], policy: MathPolicy) {
+    let prefix = if policy == MathPolicy::Sse2 {
+        values.len() / 4 * 4
+    } else {
+        0
+    };
+    let (vector, tail) = values.split_at_mut(prefix);
+    for group in vector.as_chunks_mut::<4>().0 {
+        let exponent = sse2_exp_f32x4(group.map(|v| 0.0 - v));
+        for (value, exp) in group.iter_mut().zip(exponent) {
+            *value /= 1.0 + exp;
+        }
+    }
+    for value in tail {
+        *value /= 1.0 + (-*value).exp();
+    }
+}

+pub fn softmax_exp_sum_f32_with_policy(
+    values: &mut [f32],
+    maximum: f32,
+    policy: MathPolicy,

+    let prefix = if policy == MathPolicy::Sse2 {
+        values.len() / 4 * 4
+    } else {
+        0
+    };
+    let (vector, tail) = values.split_at_mut(prefix);
+    let mut sum = 0.0_f64;

+        let output = sse2_exp_f32x4(group.map(|v| v - maximum));
+        group.copy_from_slice(&output);

+        let pairs = ordered_add_f32(

+            ordered_add_f32(output[2], output[3]),
+        );
+        sum = ordered_add_f64(sum, pairs as f64);
+    }
+    for value in tail {
+        *value = (*value - maximum).exp();
+        sum = if policy == MathPolicy::Sse2 {
+            ordered_add_f64(sum, *value as f64)
+        } else {
+            sum + *value as f64
+        };
+    }
+    sum

+/// Original four-lane SSE2 exponential. Constants are exact source F32 bits.
+/// This intentionally reproduces the original approximation and scaling, rather
+/// than replacing it with a differently rounded scalar exponential.
+pub fn sse2_exp_f32x4(input: [f32; 4]) -> [f32; 4] {
+    let parts = input.map(|x| {
+        let z = x * f32::from_bits(0x3fb8_aa3b) + f32::from_bits(0x4b40_0000);
+        let n = z - f32::from_bits(0x4b40_0000);
+        let b = (x - n * f32::from_bits(0x3f31_7200)) - n * f32::from_bits(0x35bf_be8e);
+        let exponent = z.to_bits().wrapping_shl(23);
+        let scale = f32::from_bits(exponent.wrapping_add(0x3f80_0000));
+        let squared = b * b;
+        let first = f32::from_bits(0x3c07_2010) * b + f32::from_bits(0x3d2b_9f17);
+        let second = f32::from_bits(0x3e2a_af33) * b + f32::from_bits(0x3eff_fedb);
+        let polynomial = (first * squared + second) * squared + f32::from_bits(0x3f7f_fff6) * b;
+        (n, exponent, scale, polynomial)
+    });
+    // The source branches once for the complete vector, not once per element.

+pub fn center_f32_and_variance_with_policy(
+    values: &mut [f32],
+    mean: f32,
+    policy: MathPolicy,
+) -> f64 {
+    let width = values.len();
+    let prefix = width / 4 * 4;
+    let (vector, tail) = values.split_at_mut(prefix);
+    let mut sum = 0.0_f64;
+    for group in vector.as_chunks_mut::<4>().0 {
+        for value in group.iter_mut() {
+            *value -= mean;
+        }
+        let squares = group.map(|value| value * value);
+        if policy == MathPolicy::Sse2 {
+            let paired = ordered_add_f32(
+                ordered_add_f32(squares[0], squares[1]),
+                ordered_add_f32(squares[2], squares[3]),
+            );
+            sum = ordered_add_f64(sum, paired as f64);
+        } else {
+            // The frozen MSVC scalar function unrolls four iterations. Each
+            // F64 add places the new square in the first operand; its scalar
+            // tail places the running sum first. Numeric reduction order is
+            // unchanged, but the distinction preserves observed NaN payloads.
+            for square in squares {
+                sum = ordered_add_f64(square as f64, sum);
+            }
+        }
+    }
+    for value in tail {
+        *value -= mean;
+        let squared = *value * *value;
+        sum = ordered_add_f64(sum, squared as f64);
+    }
+    sum / width as f64
+}

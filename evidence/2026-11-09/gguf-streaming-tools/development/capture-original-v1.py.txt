def u32(x,e='<'):return struct.pack(e+'I',x)
def u64(x,e='<'):return struct.pack(e+'Q',x)
def st(s,e='<'):
 b=s.encode('utf-8') if isinstance(s,str) else s
 return u64(len(b),e)+b
def val(t,v,e='<'):
 if t==8:return st(v,e)
 if t==9:
  et,vs=v
  return u32(et,e)+u64(len(vs),e)+b''.join(val(et,x,e) for x in vs)
 if isinstance(v,bytes):return v
 return struct.pack(e+{0:'B',1:'b',2:'H',3:'h',4:'I',5:'i',6:'f',7:'B',10:'Q',11:'q',12:'d'}[t],v)
def gguf(metadata,tensors,version=3,alignment=32,endian='<'):
 header=b'GGUF'+u32(version,endian)+u64(len(tensors),endian)+u64(len(metadata),endian)
 for key,t,v in metadata:header+=st(key,endian)+u32(t,endian)+val(t,v,endian)
 offset=0
 for name,dims,dtype,data in tensors:
  header+=st(name,endian)+u32(len(dims),endian)+b''.join(u64(x,endian) for x in dims)+u32(dtype,endian)+u64(offset,endian)
  offset+=(len(data)+alignment-1)//alignment*alignment
 header+=bytes((-len(header))%alignment)
 return header+b''.join(data+bytes((-len(data))%alignment) for _,_,_,data in tensors)

+// Match the defined decimal-prefix behavior of atoi/sscanf("%d"). Overflow and
+// a missing sscanf value are checked instead of reproducing C undefined behavior.
+fn decimal_prefix(text: &str) -> Result<Option<i32>, String> {
+    let bytes = text.as_bytes();
+    let mut index = 0;
+    while bytes
+        .get(index)
+        .is_some_and(|c| matches!(c, b' ' | b'\t' | b'\n' | b'\r' | b'\x0b' | b'\x0c'))
+    {
+        index += 1;
+    }
+    let negative = bytes.get(index) == Some(&b'-');
+    if negative || bytes.get(index) == Some(&b'+') {
+        index += 1;
+    }
+    let start = index;
+    let mut value = 0u64;
+    while let Some(&digit) = bytes.get(index).filter(|c| c.is_ascii_digit()) {
+        value = value
+            .checked_mul(10)
+            .and_then(|n| n.checked_add(u64::from(digit - b'0')))
+            .ok_or_else(|| {
+                "error: decimal argument overflows the source signed integer".to_string()
+            })?;
+        if value > i32::MAX as u64 + u64::from(negative) {
+            return Err("error: decimal argument overflows the source signed integer".into());
+        }
+        index += 1;
+    }
+    if index == start {
+        return Ok(None);
+    }
+    let value = if negative {
+        -(value as i64)
+    } else {
+        value as i64
+    };
+    Ok(Some(value as i32))
+}
+fn size(text: &str) -> Result<u64, String> {
+    let unit = match text.as_bytes().last() {
+        Some(b'M') => 1_000_000u64,
+        Some(b'G') => 1_000_000_000u64,
+        Some(c) => {
+            return Err(format!(
+                "error: supported units are M (megabytes) or G (gigabytes), but got: {}",
+                char::from(*c)
+            ));
+        }
+        None => return Err("error: size argument is empty".into()),
+    };
+    let n = decimal_prefix(text)?
+        .ok_or_else(|| "error: size argument has no decimal value".to_string())?;
+    if n <= 0 {
+        return Err("error: size must be a positive value".into());
+    }
+    (n as u64)
+        .checked_mul(unit)
+        .ok_or_else(|| "error: size argument overflows the byte count".into())
+}
+fn parse(args: &[OsString]) -> Result<Action, String> {
+    let mut operation = None;
+    let mut mode = None;
+    let mut split = SplitOptions::default();
+    let mut merge = MergeOptions::default();
+    let mut index = 0;
+    while let Some(raw) = args
+        .get(index)
+        .and_then(|x| x.to_str())
+        .filter(|x| x.starts_with("--"))
+    {
+        let arg = raw.replace('_', "-");
+        match arg.as_str() {
+            "--help" => return Ok(Action::Help),
+            "--version" => return Ok(Action::Version),
+            "--dry-run" => {
+                split.dry_run = true;
+                merge.dry_run = true;
+            }
+            "--no-tensor-first-split" => split.no_tensor_first_split = true,
+            "--delete-splits" => merge.delete_splits = true,
+            "--split" | "--merge" => {
+                let next = if arg == "--split" {
+                    Operation::Split
+                } else {
+                    Operation::Merge
+                };
+                if operation.is_some_and(|old| old != next) {
+                    return Err(
+                        "error: either --split or --merge can be specified, but not both".into(),
+                    );
+                }
+                operation = Some(next);
+            }
+            "--split-max-tensors" | "--split-max-size" => {
+                index += 1;
+                let value = args
+                    .get(index)
+                    .ok_or_else(|| format!("error: invalid parameter for argument: {arg}"))?;
+                let value = value
+                    .to_str()
+                    .ok_or_else(|| format!("error: non-text parameter for argument: {arg}"))?;
+                let is_size = arg == "--split-max-size";
+                if mode.is_some_and(|old| old != is_size) {
+                    return Err("error: either --split-max-tensors or --split-max-size can be specified, but not both".into());
+                }
+                mode = Some(is_size);
+                split.limit = if is_size {
+                    SplitLimit::Bytes(size(value)?)
+                } else {
+                    SplitLimit::Tensors(decimal_prefix(value)?.unwrap_or(0))
+                };
+            }
+            _ => return Err(format!("error: unknown argument: {arg}")),
+        }
+        index += 1;
+    }
+    if args.len() - index != 2 {
+        return Err("error: bad arguments".into());
+    }
+    Ok(Action::Run {
+        operation: operation.unwrap_or(Operation::Split),
+        input: PathBuf::from(&args[index]),
+        output: PathBuf::from(&args[index + 1]),
+        split,
+        merge,
+    })
+}
+fn main() -> ExitCode {
+    let mut args = std::env::args_os();
+    let executable = args
+        .next()
+        .unwrap_or_else(|| OsString::from("llama-gguf-split"));
+    let args: Vec<_> = args.collect();
+    let mut out = io::stdout().lock();
+    let mut err = io::stderr().lock();
+    let action = match parse(&args) {
+        Ok(action) => action,
+        Err(message) => {
+            let _ = writeln!(err, "{message}");
+            let _ = usage(&mut out, &executable);
+            return ExitCode::FAILURE;
+        }
+    };
+    let result: Result<(), Box<dyn std::error::Error>> = match action {
+        Action::Help => usage(&mut out, &executable).map_err(Into::into),
+        Action::Version => writeln!(
+            err,
+            "version: rust-llama {}\nnative Rust GGUF utility for {}-{}",
+            env!("CARGO_PKG_VERSION"),
+            std::env::consts::ARCH,
+            std::env::consts::OS
+        )
+        .map_err(Into::into),
+        Action::Run {
+            operation,
+            input,
+            output,
+            split,
+            merge,
+        } => {
+            let limits = ShardLimits::default();
+            match operation {
+                Operation::Split => {
+                    split::split(&input, &output, split, &limits, &mut out, &mut err)
+                }
+                Operation::Merge => split::merge(&input, &output, merge, &limits, &mut err),
+            }
+            .map(|_| ())
+            .map_err(Into::into)
+        }
+    };
+    match result {
+        Ok(()) => ExitCode::SUCCESS,
+        Err(error) => {
+            let _ = writeln!(err, "error: {error}");
+            ExitCode::FAILURE
+        }
+    }

+fn run(work: &WorkingDirectory, args: &[String]) -> Result<Captured, String> {
+    let stdout_path = work.root.join("__capture_stdout.txt");
+    let stderr_path = work.root.join("__capture_stderr.txt");
+    let mut child = ChildGuard(
+        Command::new(env!("CARGO_BIN_EXE_llama-gguf-split"))
+            .args(args)
+            .current_dir(&work.root)
+            .stdin(Stdio::null())
+            .stdout(Stdio::from(
+                File::create(&stdout_path).map_err(|e| e.to_string())?,
+            ))
+            .stderr(Stdio::from(
+                File::create(&stderr_path).map_err(|e| e.to_string())?,
+            ))
+            .spawn()
+            .map_err(|e| e.to_string())?,
+    );
+    let started = Instant::now();
+    let status = loop {
+        if let Some(status) = child.0.try_wait().map_err(|e| e.to_string())? {
+            break status;
+        }
+        if started.elapsed() >= Duration::from_secs(30) {
+            return Err("CLI exceeded its 30-second direct-child deadline".into());
+        }
+        thread::sleep(Duration::from_millis(5));
+    };
+    Ok(Captured {
+        status,
+        stdout: limited_text(&stdout_path)?,
+        stderr: limited_text(&stderr_path)?,

+fn disk_files(root: &Path) -> Result<BTreeMap<String, PathBuf>, String> {
+    fn visit(
+        root: &Path,
+        directory: &Path,
+        result: &mut BTreeMap<String, PathBuf>,
+    ) -> Result<(), String> {
+        for entry in fs::read_dir(directory).map_err(|e| e.to_string())? {
+            let entry = entry.map_err(|e| e.to_string())?;
+            let path = entry.path();
+            let kind = entry.file_type().map_err(|e| e.to_string())?;
+            if kind.is_symlink() {
+                return Err(format!("unexpected output symlink {}", path.display()));
+            }
+            if kind.is_dir() {
+                visit(root, &path, result)?;
+            } else if kind.is_file() {
+                let name = path
+                    .strip_prefix(root)
+                    .unwrap()
+                    .to_str()
+                    .unwrap()
+                    .replace('\\', "/");
+                if !matches!(
+                    name.as_str(),
+                    "__capture_stdout.txt" | "__capture_stderr.txt"
+                ) {
+                    result.insert(name, path);
+                }
+            } else {
+                return Err(format!("unexpected output file type {}", path.display()));
+            }
+        }
+        Ok(())
+    }
+    let mut result = BTreeMap::new();
+    visit(root, root, &mut result)?;
+    Ok(result)

+fn compare_file(path: &Path, expected: &[u8]) -> Result<(), String> {
+    let size = fs::metadata(path).map_err(|e| e.to_string())?.len();
+    if size != expected.len() as u64 {
+        return Err(format!(
+            "{}: {size} bytes, expected {}",
+            path.display(),
+            expected.len()
+        ));
+    }
+    let actual = fs::read(path).map_err(|e| e.to_string())?;
+    if actual != expected {
+        return Err(format!(
+            "{}: SHA {} != expected {}",
+            path.display(),
+            sha256(&actual),
+            sha256(expected)
+        ));
+    }
+    Ok(())

+fn verify_failure_files(

+    case: &Value,

+## Ordered writing and the split/merge executable
+
+`Gguf::metadata_order()` exposes the original serialized order of keys still
+present in the public map. Newly inserted map keys have no original position;
+callers constructing a modified header supply their chosen order explicitly.

+borrowed ordered key/value list and borrowed payloads. `write_header` accepts
+borrowed `TensorDescriptor` values and explicit `Limits`, writes a little-endian

+It validates the entire header and tensor layout before its first output byte.

+the beginning of its output and then streams the payloads using those offsets.

+    while bytes != 0 {
+        let length = bytes.min(buffer.len() as u64) as usize;
+        input.read_exact(&mut buffer[..length])?;
+        out.write_all(&buffer[..length])?;
+        bytes -= length as u64;
+    }
+    Ok(())
+}
+fn reject_same_path(input: &Path, output: &Path) -> Result<()> {
+    if output.exists() && fs::canonicalize(input)? == fs::canonicalize(output)? {
+        return Err(invalid("output aliases an input file"));
+    }
+    #[cfg(unix)]
+    if output.exists() {
+        use std::os::unix::fs::MetadataExt;
+        let a = fs::metadata(input)?;
+        let b = fs::metadata(output)?;
+        if a.dev() == b.dev() && a.ino() == b.ino() {
+            return Err(invalid("output is a hard link to an input file"));
+        }
+    }
+    Ok(())
+}
+fn open_split_output(input: &File, path: &Path) -> Result<File> {
+    // Opening without truncation retains the exact destination handle across
+    // the identity check. A symlink/path swap after the earlier preflight
+    // cannot turn set_len or subsequent writes into input-file truncation.
+    let destination = OpenOptions::new()
+        .write(true)
+        .create(true)
+        .truncate(false)
+        .open(path)?;
+    #[cfg(unix)]
+    {
+        use std::os::unix::fs::MetadataExt;
+        let source = input.metadata()?;
+        let target = destination.metadata()?;
+        if source.dev() == target.dev() && source.ino() == target.ino() {
+            return Err(invalid("output handle aliases the retained input file"));
+        }
+    }
+    #[cfg(not(unix))]
+    let _ = input;
+    // Windows inputs deny write sharing while retained, so opening an alias
+    // above fails before any truncation. See original_header.
+    destination.set_len(0)?;
+    Ok(destination)
+}
+pub fn split_path(prefix: &Path, index: usize, count: usize) -> Result<PathBuf> {
+    if count == 0 || count > u16::MAX as usize || index >= count {
+        return Err(invalid("split index/count outside UINT16 range"));
+    }
+    let mut name = prefix.as_os_str().to_os_string();
+    name.push(format!("-{:05}-of-{count:05}.gguf", index + 1));
+    Ok(PathBuf::from(name))
+}
+fn split_prefix(path: &Path, count: usize) -> Result<PathBuf> {
+    let suffix = format!("-00001-of-{count:05}.gguf");
+    #[cfg(unix)]
+    let prefix = {
+        use std::os::unix::ffi::{OsStrExt, OsStringExt};
+        let bytes = path.as_os_str().as_bytes();
+        let prefix = bytes
+            .strip_suffix(suffix.as_bytes())
+            .ok_or_else(|| invalid("unexpected first split filename"))?;
+        OsString::from_vec(prefix.to_vec())
+    };
+    #[cfg(windows)]
+    let prefix = {
+        use std::os::windows::ffi::{OsStrExt, OsStringExt};
+        let bytes: Vec<u16> = path.as_os_str().encode_wide().collect();
+        let suffix: Vec<u16> = suffix.encode_utf16().collect();
+        let prefix = bytes
+            .strip_suffix(suffix.as_slice())
+            .ok_or_else(|| invalid("unexpected first split filename"))?;
+        OsString::from_wide(prefix)
+    };
+    #[cfg(not(any(unix, windows)))]
+    let prefix = {
+        let path = path
+            .to_str()
+            .ok_or_else(|| Error::Unsupported("native path encoding".into()))?;
+        OsString::from(
+            path.strip_suffix(&suffix)
+                .ok_or_else(|| invalid("unexpected first split filename"))?,
+        )
+    };
+    if prefix.is_empty() {
+        return Err(invalid("split filename has an empty prefix"));
+    }
+    Ok(PathBuf::from(prefix))
+}

+/// Split one GGUF. The first output contains primary metadata; later files
+/// contain only the three split keys. Dry-run validates/plans headers without
+/// requiring complete tensor payloads, as in the original tool. Bounded header
+/// read-ahead can still prefetch a payload prefix.
+pub fn split<W: Write, E: Write>(
+    input: &Path,
+    output_prefix: &Path,
+    options: SplitOptions,
+    limits: &ShardLimits,
+    stdout: &mut W,
+    stderr: &mut E,
+) -> Result<Summary> {
+    let mut file = original_header(input, limits.gguf.clone())?;
+    let count = file.tensors.len();
+    if count > i32::MAX as usize {
+        return Err(invalid("tensor count exceeds source INT32 field"));
+    }
+    match options.limit {
+        SplitLimit::Tensors(0) if count > 1 => {
+            return Err(invalid("zero tensor split limit would divide by zero"));
+        }
+        SplitLimit::Bytes(0) => return Err(invalid("size must be a positive value")),
+        _ => {}
+    }
+    let max_shards = limits.max_shards.min(u16::MAX as usize);
+    if max_shards == 0 {
+        return Err(Error::Limit("split file budget is zero".into()));
+    }
+    let mut ranges = Vec::new();
+    ranges
+        .try_reserve_exact((count + 2).min(max_shards))
+        .map_err(|_| Error::Limit("split plan allocation".into()))?;
+    if options.no_tensor_first_split {
+        ranges.push(0..0);
+    }
+    let mut start = 0;
+    let mut bytes = 0u64;
+    for (index, tensor) in file.tensors.iter().enumerate() {
+        let padded = align(tensor.byte_len, 32)?;
+        let next = bytes
+            .checked_add(padded)
+            .ok_or_else(|| invalid("split byte count overflow"))?;
+        let boundary = match options.limit {
+            SplitLimit::Tensors(number) => index > 0 && index % number.unsigned_abs() as usize == 0,
+            SplitLimit::Bytes(limit) => next > limit,
+        };
+        if boundary {
+            if index == start {
+                return Err(invalid(
+                    "one split would contain zero tensors; size limit is too small",
+                ));
+            }
+            if ranges.len() + 1 >= max_shards {
+                return Err(Error::Limit("split count exceeds file budget".into()));
+            }
+            ranges.push(start..index);
+            start = index;
+            bytes = padded;
+        } else {
+            bytes = next;
+        }
+    }
+    if ranges.len() >= max_shards {
+        return Err(Error::Limit("split count exceeds file budget".into()));
+    }
+    ranges.push(start..count);
+    // Every output header and path is checked before the first file is opened.
+    let mut plans = Vec::new();
+    plans
+        .try_reserve_exact(ranges.len())
+        .map_err(|_| Error::Limit("split output plan allocation".into()))?;
+    for (index, range) in ranges.iter().enumerate() {
+        let header = split_header(
+            &file,
+            range.clone(),
+            index,
+            ranges.len(),
+            &limits.gguf,
+            &mut io::sink(),
+        )?;
+        let path = split_path(output_prefix, index, ranges.len())?;
+        reject_same_path(input, &path)?;
+        let raw_bytes = file.tensors[range.clone()]
+            .iter()
+            .try_fold(0u64, |sum, tensor| {
+                sum.checked_add(tensor.byte_len)
+                    .ok_or_else(|| invalid("display byte count overflow"))
+            })?;
+        let displayed_mb = header
+            .data_offset
+            .checked_add(raw_bytes)
+            .ok_or_else(|| invalid("display size overflow"))?
+            / 1_000_000;
+        plans.push((path, header, displayed_mb));
+    }
+    writeln!(stdout, "n_split: {}", ranges.len())?;
+    for (index, (range, (_, _, size))) in ranges.iter().zip(&plans).enumerate() {
+        writeln!(
+            stdout,
+            "split {:05}: n_tensors = {}, total_size = {size}M",
+            index + 1,
+            range.len()
+        )?;
+    }
+    if !options.dry_run {
+        let mut copy_buffer = buffer(
+            &limits.gguf,
+            file.tensors.iter().any(|tensor| tensor.byte_len != 0),
+        )?;
+        for (index, (range, (path, expected, _))) in ranges.iter().zip(&plans).enumerate() {
+            write!(stdout, "Writing file {} ... ", path.display())?;
+            stdout.flush()?;
+            let destination = open_split_output(&file.reader, path)?;
+            let mut output = BufWriter::with_capacity(64 * 1024, destination);
+            let actual = split_header(
+                &file,
+                range.clone(),
+                index,
+                ranges.len(),
+                &limits.gguf,
+                &mut output,
+            )?;
+            if &actual != expected {
+                return Err(invalid("split plan changed before writing"));
+            }
+            for tensor_index in range.clone() {
+                let tensor = &file.tensors[tensor_index];
+                let offset = file
+                    .data_offset
+                    .checked_add(tensor.offset)
+                    .ok_or_else(|| invalid("input offset overflow"))?;
+                copy_exact(
+                    &mut file.reader,
+                    &mut output,
+                    offset,
+                    tensor.byte_len,
+                    &mut copy_buffer,
+                )?;
+                writer::zeros(&mut output, align(tensor.byte_len, 32)? - tensor.byte_len)?;
+            }
+            output.flush()?;
+            writeln!(stdout, "done")?;
+        }
+    }
+    writeln!(
+        stderr,
+        "gguf_split: {} gguf split written with a total of {count} tensors.",
+        ranges.len()
+    )?;
+    Ok(Summary {
+        shards: ranges.len(),
+        tensors: count,
+    })
+}
+fn remaining_limits(limits: &Limits, used: HeaderUsage) -> Result<Limits> {
+    let mut remaining = limits.clone();
+    remaining.max_metadata_bytes = limits
+        .max_metadata_bytes
+        .checked_sub(used.bytes)
+        .ok_or_else(|| Error::Limit("combined header byte budget".into()))?;
+    remaining.max_metadata_entries = limits
+        .max_metadata_entries
+        .checked_sub(used.metadata_entries)
+        .ok_or_else(|| Error::Limit("combined metadata entry budget".into()))?;
+    remaining.max_total_values = limits
+        .max_total_values
+        .checked_sub(used.metadata_values)
+        .ok_or_else(|| Error::Limit("combined metadata value budget".into()))?;
+    remaining.max_tensors = limits
+        .max_tensors
+        .checked_sub(used.tensors)
+        .ok_or_else(|| Error::Limit("combined tensor budget".into()))?;
+    Ok(remaining)

+fn charge(used: &mut HeaderUsage, new: HeaderUsage) -> Result<()> {
+    used.bytes = used
+        .bytes
+        .checked_add(new.bytes)
+        .ok_or_else(|| invalid("combined header bytes overflow"))?;
+    used.metadata_entries = used
+        .metadata_entries
+        .checked_add(new.metadata_entries)
+        .ok_or_else(|| invalid("combined metadata entries overflow"))?;
+    used.metadata_values = used
+        .metadata_values
+        .checked_add(new.metadata_values)
+        .ok_or_else(|| invalid("combined metadata values overflow"))?;
+    used.tensors = used
+        .tensors
+        .checked_add(new.tensors)
+        .ok_or_else(|| invalid("combined tensor count overflow"))?;
+    Ok(())

+fn merge_header(
+    files: &[Gguf<File>],
+    limits: &Limits,
+    out: &mut impl Write,
+) -> Result<WriteLayout> {
+    let first = files.first().ok_or_else(|| invalid("empty merge input"))?;
+    let count = Value::U16(0);
+    let mut metadata = Vec::new();
+    metadata
+        .try_reserve_exact(first.metadata.len())
+        .map_err(|_| Error::Limit("merge metadata index allocation".into()))?;
+    for key in first.metadata_order() {
+        if key != "split.count" {
+            metadata.push((key, &first.metadata[key]));
+        }
+    }
+    metadata.push(("split.count", &count));
+    let total = files.iter().try_fold(0usize, |sum, file| {
+        sum.checked_add(file.tensors.len())
+            .ok_or_else(|| invalid("merged tensor count overflow"))
+    })?;
+    let mut tensors = Vec::new();
+    tensors
+        .try_reserve_exact(total)
+        .map_err(|_| Error::Limit("merge descriptor allocation".into()))?;
+    for file in files {
+        tensors.extend(descriptors(file, 0..file.tensors.len())?);
+    }
+    writer::header_with_policy(out, &metadata, &tensors, limits, HeaderPolicy::FROZEN_TOOL)

+/// Merge numbered shards. Primary split.no is deliberately not consulted;
+/// subsidiary split.no and split.count must match, and the primary signed
+/// tensor total must match the complete directory. This differs from model
+/// shard discovery and therefore does not reuse its acceptance policy.
+///
+/// Existing output files are refused even during dry-run. With delete_splits,
+/// source deletion starts only after the complete output is flushed and synced.
+/// This strengthens the original tool's failure safety; successful bytes and
+/// deletions are the same. Failures before a complete flushed/synced output
+/// preserve every input. Cleanup can fail after removing some shards; the
+/// completed output remains. Destructive cleanup requires stable, exclusively
+/// managed paths: standard Rust has no atomic identity-check-and-unlink API.
+pub fn merge<W: Write>(
+    input: &Path,
+    output: &Path,
+    options: MergeOptions,
+    limits: &ShardLimits,
+    stderr: &mut W,
+) -> Result<Summary> {
+    writeln!(
+        stderr,
+        "gguf_merge: {} -> {}",
+        input.display(),
+        output.display()
+    )?;
+    if output.exists() {
+        return Err(invalid(format!(
+            "output file {} already exists",
+            output.display()
+        )));
+    }
+    let first = original_header(input, limits.gguf.clone())?;
+    let count = match first.metadata.get("split.count") {
+        Some(Value::U16(count)) if *count != 0 => usize::from(*count),
+        _ => return Err(invalid("input requires a positive UINT16 split.count")),
+    };
+    if count > limits.max_shards {
+        return Err(Error::Limit("merge exceeds owned file budget".into()));
+    }
+    let prefix = split_prefix(input, count)?;
+    let mut used = first.header_usage();
+    let mut files = Vec::new();
+    let mut paths = Vec::new();
+    files
+        .try_reserve_exact(count)
+        .map_err(|_| Error::Limit("merge file owner allocation".into()))?;
+    paths
+        .try_reserve_exact(count)
+        .map_err(|_| Error::Limit("merge path allocation".into()))?;
+    write!(
+        stderr,
+        "gguf_merge: reading metadata {} ...",
+        input.display()
+    )?;
+    files.push(first);
+    paths.push(input.to_path_buf());
+    writeln!(stderr, "\x1b[3Ddone")?;
+    for index in 1..count {
+        let path = split_path(&prefix, index, count)?;
+        write!(
+            stderr,
+            "gguf_merge: reading metadata {} ...",
+            path.display()
+        )?;
+        let file = original_header(&path, remaining_limits(&limits.gguf, used)?)?;
+        if !matches!(file.metadata.get("split.no"), Some(Value::U16(number)) if usize::from(*number)==index)
+            || !matches!(file.metadata.get("split.count"), Some(Value::U16(number)) if usize::from(*number)==count)
+        {
+            return Err(invalid(format!(
+                "{} is not split {} of {count}",
+                path.display(),
+                index + 1
+            )));
+        }
+        charge(&mut used, file.header_usage())?;
+        files.push(file);
+        paths.push(path);
+        writeln!(stderr, "\x1b[3Ddone")?;
+    }
+    if !matches!(files[0].metadata.get("split.tensors.count"), Some(Value::I32(total)) if *total >= 0 && *total as usize == used.tensors)
+    {
+        return Err(invalid(
+            "split tensor total does not match primary INT32 split.tensors.count",
+        ));
+    }
+    let plan = merge_header(&files, &limits.gguf, &mut io::sink())?;
+    let mut copy_buffer = buffer(
+        &limits.gguf,
+        files
+            .iter()
+            .any(|file| file.tensors.iter().any(|tensor| tensor.byte_len != 0)),
+    )?;
+    let mut destination = if options.dry_run {
+        None
+    } else {
+        let file = OpenOptions::new()
+            .write(true)
+            .create_new(true)
+            .open(output)?;
+        let mut out = BufWriter::with_capacity(64 * 1024, file);
+        writer::zeros(&mut out, plan.data_offset)?;
+        Some(out)
+    };
+    for (file, path) in files.iter_mut().zip(&paths) {
+        write!(stderr, "gguf_merge: writing tensors {} ...", path.display())?;
+        for tensor in &file.tensors {
+            let offset = file
+                .data_offset
+                .checked_add(tensor.offset)
+                .ok_or_else(|| invalid("input tensor offset overflow"))?;
+            if let Some(out) = &mut destination {
+                copy_exact(
+                    &mut file.reader,
+                    out,
+                    offset,
+                    tensor.byte_len,
+                    &mut copy_buffer,
+                )?;
+                writer::zeros(out, align(tensor.byte_len, 32)? - tensor.byte_len)?;
+            } else {
+                copy_exact(
+                    &mut file.reader,
+                    &mut io::sink(),
+                    offset,
+                    tensor.byte_len,
+                    &mut copy_buffer,
+                )?;
+            }
+        }
+        writeln!(stderr, "\x1b[3Ddone")?;
+    }
+    if let Some(out) = &mut destination {
+        out.seek(SeekFrom::Start(0))?;
+        let actual = merge_header(&files, &limits.gguf, out)?;
+        if actual != plan {
+            return Err(invalid("merge plan changed before writing"));
+        }
+        out.flush()?;
+        if options.delete_splits {
+            out.get_ref().sync_all()?;
+        }
+    }
+    // Close all input/output handles before optional deletion (also on Windows).
+    drop(files);
+    drop(destination);
+    let mut deletion_error = None;
+    if !options.dry_run && options.delete_splits {
+        for path in &paths {
+            match fs::remove_file(path) {
+                Ok(()) => writeln!(stderr, "gguf_merge: deleted file {}", path.display())?,
+                Err(error) => {
+                    writeln!(stderr, "error: failed to delete {}", path.display())?;
+                    if deletion_error.is_none() {
+                        deletion_error = Some(error);
+                    }
+                }
+            }
+        }
+    }
+    writeln!(
+        stderr,
+        "gguf_merge: {} merged from {count} split with {} tensors.",
+        output.display(),
+        used.tensors
+    )?;
+    if let Some(error) = deletion_error {
+        return Err(Error::Io(error));
+    }
+    Ok(Summary {
+        shards: count,
+        tensors: used.tensors,
+    })
+}

+fn value_size(
+    value: &Value,
+    limits: &Limits,
+    policy: HeaderPolicy,
+    total_values: &mut usize,
+) -> Result<u64> {
+    *total_values = total_values
+        .checked_add(1)
+        .filter(|count| *count <= limits.max_total_values)
+        .ok_or_else(|| Error::Limit("encoded metadata value budget".into()))?;
+    Ok(match value {
+        Value::U8(_) | Value::I8(_) | Value::Bool(_) => 1,
+        Value::U16(_) | Value::I16(_) => 2,
+        Value::U32(_) | Value::I32(_) | Value::F32(_) => 4,
+        Value::U64(_) | Value::I64(_) | Value::F64(_) => 8,
+        Value::String(text) => string_size(text, limits, policy)?,
+        Value::Array {
+            element_type,
+            values,
+        } => {
+            if *element_type > 12 || *element_type == 9 {
+                return Err(invalid("heterogeneous or nested array"));
+            }
+            if values.len() > limits.max_array_values
+                || values.len() > limits.max_total_values.saturating_sub(*total_values)
+            {
+                return Err(Error::Limit("encoded array value budget".into()));
+            }
+            let mut bytes = 12;
+            for item in values {
+                if item.type_id() != *element_type {
+                    return Err(invalid("heterogeneous or nested array"));
+                }
+                add(
+                    &mut bytes,
+                    value_size(item, limits, policy, total_values)?,
+                    limits.max_metadata_bytes,
+                )?;
+            }
+            bytes
+        }
+    })

+        ));
+    }
+    let metadata_alignment = match metadata.iter().find(|(key, _)| *key == "general.alignment") {
+        None => 32,
+        Some((_, Value::U32(value))) => *value,
+        _ => return Err(invalid("general.alignment must be UINT32")),
+    };
+    if !metadata_alignment.is_power_of_two() || metadata_alignment > 1024 * 1024 {
+        return Err(invalid("invalid alignment"));
+    }
+    let alignment = if policy.frozen_tool {
+        32
+    } else {
+        metadata_alignment
+    };
+    let mut header = 0;
+    add(&mut header, 24, limits.max_metadata_bytes)?;
+    let mut keys = BTreeSet::new();
+    let mut values = 0;
+    for &(key, value) in metadata {
+        if key.is_empty() || key.contains('\0') || !keys.insert(key) {
+            return Err(invalid("empty, NUL or duplicated metadata key"));
+        }
+        add(
+            &mut header,
+            string_size(key, limits, policy)?,
+            limits.max_metadata_bytes,
+        )?;
+        add(&mut header, 4, limits.max_metadata_bytes)?;
+        add(
+            &mut header,
+            value_size(value, limits, policy, &mut values)?,
+            limits.max_metadata_bytes,
+        )?;
+    }
+    // A rank-zero tensor with an empty name still needs 24 encoded bytes.
+    // Reject impossible descriptor counts before reserving per-tensor arrays.
+    let minimum = u64::try_from(tensors.len())
+        .ok()
+        .and_then(|count| count.checked_mul(24))
+        .ok_or_else(|| Error::Limit("encoded tensor descriptor count overflow".into()))?;
+    if minimum > limits.max_metadata_bytes.saturating_sub(header) {
+        return Err(Error::Limit(
+            "tensor descriptors minimum encoded size exceeds remaining metadata budget".into(),
+        ));
+    }
+    #[cfg(test)]
+    LAYOUT_RESERVATIONS.with(|calls| calls.set(calls.get() + 1));
+    let mut tensor_offsets = Vec::new();
+    let mut lengths = Vec::new();
+    tensor_offsets
+        .try_reserve_exact(tensors.len())
+        .map_err(|_| Error::Limit("writer tensor offset allocation".into()))?;
+    lengths
+        .try_reserve_exact(tensors.len())
+        .map_err(|_| Error::Limit("writer tensor length allocation".into()))?;
+    let mut names = BTreeSet::new();
+    let mut offset = 0u64;
+    for tensor in tensors {
+        if tensor.name.len() >= 64 || tensor.name.contains('\0') || !names.insert(tensor.name) {
+            return Err(invalid("invalid or duplicated tensor name"));
+        }
+        let bytes = tensor_bytes(tensor)?;
+        let rank = policy.rank(tensor.dimensions);
+        add(
+            &mut header,
+            string_size(tensor.name, limits, policy)?,
+            limits.max_metadata_bytes,
+        )?;
+        add(&mut header, 16 + rank as u64 * 8, limits.max_metadata_bytes)?;
+        tensor_offsets.push(offset);
+        lengths.push(bytes);
+        offset = offset
+            .checked_add(align(bytes, alignment as u64)?)
+            .ok_or_else(|| invalid("tensor offset overflow"))?;
+    }
+    let data_offset = align(header, alignment as u64)?;
+    if data_offset > limits.max_metadata_bytes {
+        return Err(Error::Limit(
+            "encoded header including padding exceeds budget".into(),
+        ));
+    }
+    let file_bytes = data_offset
+        .checked_add(offset)
+        .ok_or_else(|| invalid("output file size overflow"))?;
+    Ok(WriteLayout {
+        data_offset,
+        alignment,
+        tensor_offsets,
+        tensor_bytes: lengths,
+        file_bytes,
+    })
+}

+pub(crate) fn zeros<W: Write>(out: &mut W, mut count: u64) -> Result<()> {
+    const ZERO: [u8; 4096] = [0; 4096];
+    while count != 0 {
+        let part = count.min(ZERO.len() as u64) as usize;
+        out.write_all(&ZERO[..part])?;
+        count -= part as u64;
+    }
+    Ok(())
+}
+struct Encoder<'a, W: Write> {
+    out: &'a mut W,
+    bytes: u64,
+    policy: HeaderPolicy,
+}
+impl<W: Write> Encoder<'_, W> {
+    fn raw(&mut self, bytes: &[u8]) -> Result<()> {
+        self.out.write_all(bytes)?;
+        self.bytes = self
+            .bytes
+            .checked_add(bytes.len() as u64)
+            .ok_or_else(|| invalid("encoded byte count overflow"))?;
+        Ok(())
+    }
+    fn string(&mut self, value: &str) -> Result<()> {
+        let value = self.policy.string(value);
+        self.raw(&(value.len() as u64).to_le_bytes())?;
+        self.raw(value.as_bytes())
+    }
+    fn value(&mut self, value: &Value) -> Result<()> {
+        match value {
+            Value::U8(value) => self.raw(&[*value]),
+            Value::I8(value) => self.raw(&[*value as u8]),
+            Value::U16(value) => self.raw(&value.to_le_bytes()),
+            Value::I16(value) => self.raw(&value.to_le_bytes()),
+            Value::U32(value) => self.raw(&value.to_le_bytes()),
+            Value::I32(value) => self.raw(&value.to_le_bytes()),
+            Value::F32(value) => self.raw(&value.to_le_bytes()),
+            Value::Bool(value) => self.raw(&[u8::from(*value)]),
+            Value::String(value) => self.string(value),
+            Value::U64(value) => self.raw(&value.to_le_bytes()),
+            Value::I64(value) => self.raw(&value.to_le_bytes()),
+            Value::F64(value) => self.raw(&value.to_le_bytes()),
+            Value::Array {
+                element_type,
+                values,
+            } => {
+                self.raw(&element_type.to_le_bytes())?;
+                self.raw(&(values.len() as u64).to_le_bytes())?;
+                for value in values {
+                    self.value(value)?;
+                }
+                Ok(())
+            }
+        }
+    }
+}
+pub(crate) fn header_with_policy<W: Write>(
+    out: &mut W,
+    metadata: &[(&str, &Value)],
+    tensors: &[TensorDescriptor<'_>],
+    limits: &Limits,
+    policy: HeaderPolicy,
+) -> Result<WriteLayout> {
+    let layout = layout(metadata, tensors, limits, policy)?;
+    let mut encoder = Encoder {
+        out,
+        bytes: 0,
+        policy,
+    };
+    encoder.raw(b"GGUF")?;
+    encoder.raw(&3u32.to_le_bytes())?;
+    encoder.raw(&(tensors.len() as u64).to_le_bytes())?;
+    encoder.raw(&(metadata.len() as u64).to_le_bytes())?;
+    for &(key, value) in metadata {
+        encoder.string(key)?;
+        encoder.raw(&value.type_id().to_le_bytes())?;
+        encoder.value(value)?;
+    }
+    for (tensor, &offset) in tensors.iter().zip(&layout.tensor_offsets) {
+        encoder.string(tensor.name)?;
+        let rank = policy.rank(tensor.dimensions);
+        encoder.raw(&(rank as u32).to_le_bytes())?;
+        for dimension in 0..rank {
+            encoder.raw(
+                &tensor
+                    .dimensions
+                    .get(dimension)
+                    .copied()
+                    .unwrap_or(1)
+                    .to_le_bytes(),
+            )?;
+        }
+        encoder.raw(&tensor.dtype.id().to_le_bytes())?;
+        encoder.raw(&offset.to_le_bytes())?;
+    }
+    let padding = layout
+        .data_offset
+        .checked_sub(encoder.bytes)
+        .ok_or_else(|| invalid("header preflight and encoding disagree"))?;
+    zeros(encoder.out, padding)?;
+    Ok(layout)
+}

+/// Write a little-endian v3 header in the caller's metadata order, without
+/// allocating a header-sized buffer or copying metadata arrays. The returned
+/// offsets describe the payloads and their zero padding. Validation completes
+/// before the first write; an output I/O failure can still leave partial bytes.
+pub fn write_header<W: Write>(
+    out: &mut W,
+    metadata: &[(&str, &Value)],
+    tensors: &[TensorDescriptor<'_>],
+    limits: &Limits,
+) -> Result<WriteLayout> {
+    header_with_policy(out, metadata, tensors, limits, HeaderPolicy::STANDARD)
+}

+/// Ordered counterpart to the existing in-memory payload writer.
+pub fn write_ordered<W: Write>(
+    mut out: W,
+    metadata: &[(&str, &Value)],
+    tensors: &[TensorData<'_>],
+) -> Result<()> {
+    let limits = Limits::default();
+    if tensors.len() > limits.max_tensors {
+        return Err(Error::Limit("encoded tensor count budget".into()));
+    }
+    let mut descriptors = Vec::new();
+    descriptors
+        .try_reserve_exact(tensors.len())
+        .map_err(|_| Error::Limit("writer tensor descriptor allocation".into()))?;
+    for tensor in tensors {
+        let descriptor = TensorDescriptor {
+            name: tensor.name,
+            dimensions: tensor.dimensions,
+            dtype: tensor.dtype,
+        };
+        if tensor_bytes(&descriptor)? != tensor.data.len() as u64 {
+            return Err(invalid(format!(
+                "tensor {} payload length mismatch",
+                tensor.name
+            )));
+        }
+        descriptors.push(descriptor);
+    }
+    let layout = write_header(&mut out, metadata, &descriptors, &limits)?;
+    for (tensor, &bytes) in tensors.iter().zip(&layout.tensor_bytes) {
+        out.write_all(tensor.data)?;
+        zeros(&mut out, align(bytes, layout.alignment as u64)? - bytes)?;
+    }
+    Ok(())
+}
+pub(crate) fn write_map<W: Write>(
+    out: W,
+    metadata: &BTreeMap<String, Value>,
+    tensors: &[TensorData<'_>],
+) -> Result<()> {
+    if metadata.len() > Limits::default().max_metadata_entries {
+        return Err(Error::Limit("encoded metadata entry budget".into()));
+    }
+    let mut ordered = Vec::new();
+    ordered
+        .try_reserve_exact(metadata.len())
+        .map_err(|_| Error::Limit("writer metadata index allocation".into()))?;
+    ordered.extend(metadata.iter().map(|(key, value)| (key.as_str(), value)));
+    write_ordered(out, &ordered, tensors)
+}

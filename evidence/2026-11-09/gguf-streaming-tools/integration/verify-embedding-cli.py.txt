def request(base,path,body=None):
 data=None if body is None else json.dumps(body).encode()
 req=urllib.request.Request(base+path,data=data,headers={'Content-Type':'application/json'})
 try:
  with urllib.request.urlopen(req,timeout=10) as r:return r.status,json.load(r)
 except urllib.error.HTTPError as e:return e.code,json.load(e)
def compare(actual,expected):
 if isinstance(expected,(float,int)) and not isinstance(expected,bool):
  assert isinstance(actual,(float,int)),(actual,expected)
  bound=0 # Both verified profiles require exact corresponding native JSON values.
  assert abs(actual-expected)<=bound,(actual,expected,bound)
 elif isinstance(expected,dict):
  assert set(actual)==set(expected),(set(actual),set(expected))
  for k in expected:compare(actual[k],expected[k])
 elif isinstance(expected,list):
  assert len(actual)==len(expected)
  for a,e in zip(actual,expected):compare(a,e)
 else:assert actual==expected,(actual,expected)
def start(args,name,eof=False):
 log=(out/('embedding-cli-'+name+'.log')).open('wb')
 p=subprocess.Popen([str(exe),*args],cwd=repo,stdin=subprocess.DEVNULL if eof else subprocess.PIPE,stdout=subprocess.PIPE,stderr=log)
 q=queue.Queue()
 def reader():
  line=p.stdout.readline()
  log.write(line);log.flush();q.put(line.decode('utf-8','replace'))
 threading.Thread(target=reader,daemon=True).start()
 try:
  line=q.get(timeout=15)
  assert line.startswith('Listening on http://'),line
  base=line.split(';',1)[0].removeprefix('Listening on ')
  return p,base,log
 except BaseException:
  if p.poll() is None:p.terminate()
  p.wait(timeout=10);log.close();raise
def stop(p,log,graceful=True):
 try:
  if graceful and p.poll() is None:
   p.stdin.write(b'quit\n');p.stdin.flush()
   assert p.wait(timeout=10)==0
  elif p.poll() is None:
   p.terminate();p.wait(timeout=10)
 finally:
  if p.poll() is None:p.kill();p.wait(timeout=10)
  log.close()

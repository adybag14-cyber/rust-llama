+//! Dense Qwen3 graph translated from the frozen src/models/qwen3.cpp.
+//! Independent attention width and learned per-head Q/K RMS precede full-head NeoX RoPE.
+//! The bounded causal LM slice uses the frozen scalar CPU arithmetic profiles.
+use crate::llama::KvLayer;
+use crate::{AdapterSet, Error, Fingerprint, Matrix, Result, check_epsilon, check_positive, math};

+#[derive(Debug, Clone)]
+pub struct Qwen3Config {
+    pub vocabulary: usize,
+    pub width: usize,
+    pub feed_forward: usize,
+    pub heads: usize,
+    /// Key/value width per head; independent of embedding width divided by heads.
+    pub head_dim: usize,
+    pub kv_heads: usize,
+    pub max_context: usize,
+    pub rms_epsilon: f32,
+    pub rope_base: f32,
+}
+impl Qwen3Config {
+    pub fn validate(&self) -> Result<()> {
+        check_positive(&[
+            ("vocabulary", self.vocabulary),
+            ("width", self.width),
+            ("feed_forward", self.feed_forward),
+            ("heads", self.heads),
+            ("head_dim", self.head_dim),
+            ("kv_heads", self.kv_heads),
+            ("max_context", self.max_context),
+        ])?;
+        check_epsilon(self.rms_epsilon)?;
+        if !self.head_dim()?.is_multiple_of(2) {
+            return Err(Error::InvalidConfig(
+                "Qwen3 full-head NeoX RoPE requires an even head dimension".into(),
+            ));
+        }
+        if !self.rope_base.is_finite() || self.rope_base <= 0.0 {
+            return Err(Error::InvalidConfig(
+                "RoPE base must be finite and positive".into(),
+            ));
+        }
+        self.kv_width()?
+            .checked_mul(self.max_context)
+            .ok_or_else(|| Error::InvalidConfig("Qwen3 KV size overflow".into()))?;
+        self.qkv_width()?
+            .checked_mul(std::mem::size_of::<f32>())
+            .filter(|bytes| *bytes <= isize::MAX as usize)
+            .ok_or_else(|| Error::InvalidConfig("Qwen3 projection byte size overflow".into()))?;
+        Ok(())
+    }
+    pub fn head_dim(&self) -> Result<usize> {
+        check_positive(&[("head_dim", self.head_dim)])?;
+        Ok(self.head_dim)
+    }
+    pub fn query_width(&self) -> Result<usize> {
+        check_positive(&[("heads", self.heads)])?;
+        self.heads
+            .checked_mul(self.head_dim()?)
+            .ok_or_else(|| Error::InvalidConfig("Qwen3 query width overflow".into()))
+    }
+    pub fn kv_width(&self) -> Result<usize> {
+        check_positive(&[("heads", self.heads), ("kv_heads", self.kv_heads)])?;
+        if !self.heads.is_multiple_of(self.kv_heads) {
+            return Err(Error::InvalidConfig(
+                "Qwen3 attention heads must divide evenly into KV groups".into(),
+            ));
+        }
+        self.kv_heads
+            .checked_mul(self.head_dim()?)
+            .ok_or_else(|| Error::InvalidConfig("Qwen3 KV width overflow".into()))
+    }
+    pub fn qkv_width(&self) -> Result<usize> {
+        let query = self.query_width()?;
+        self.kv_width()?
+            .checked_mul(2)
+            .and_then(|n| n.checked_add(query))
+            .ok_or_else(|| Error::InvalidConfig("Qwen3 QKV size overflow".into()))
+    }
+}

+#[derive(Debug, Clone)]
+pub enum Qwen3Qkv {
+    Separate {
+        query: Matrix,
+        key: Matrix,
+        value: Matrix,
+        query_bias: Option<Vec<f32>>,
+        key_bias: Option<Vec<f32>>,
+        value_bias: Option<Vec<f32>>,
+    },
+    /// The original loader prefers a fused bias when present. Separate biases
+    /// are applied only when the fused bias is absent and all three exist.
+    Fused {
+        projection: Matrix,
+        bias: Option<Vec<f32>>,
+        query_bias: Option<Vec<f32>>,
+        key_bias: Option<Vec<f32>>,
+        value_bias: Option<Vec<f32>>,
+    },
+}
+impl Qwen3Qkv {
+    fn validate(&self, c: &Qwen3Config) -> Result<()> {
+        let kv = c.kv_width()?;
+        let (q_bias, k_bias, v_bias) = match self {
+            Self::Separate {
+                query,
+                key,
+                value,
+                query_bias,
+                key_bias,
+                value_bias,
+            } => {
+                query.shape(c.query_width()?, c.width, "Qwen3 query")?;
+                key.shape(kv, c.width, "Qwen3 key")?;
+                value.shape(kv, c.width, "Qwen3 value")?;
+                (query_bias, key_bias, value_bias)
+            }
+            Self::Fused {
+                projection,
+                bias,
+                query_bias,
+                key_bias,
+                value_bias,
+            } => {
+                projection.shape(c.qkv_width()?, c.width, "Qwen3 fused QKV")?;
+                math::optional_bias(bias, c.qkv_width()?, "Qwen3 fused QKV bias")?;
+                if bias.is_some()
+                    && (query_bias.is_some() || key_bias.is_some() || value_bias.is_some())
+                {
+                    return Err(Error::InvalidConfig(
+                        "Qwen3 fused and separate QKV biases cannot both be active".into(),
+                    ));
+                }
+                (query_bias, key_bias, value_bias)
+            }
+        };
+        math::optional_bias(q_bias, c.query_width()?, "Qwen3 query bias")?;
+        math::optional_bias(k_bias, kv, "Qwen3 key bias")?;
+        math::optional_bias(v_bias, kv, "Qwen3 value bias")?;
+        Ok(())
+    }
+    fn project(
+        &self,
+        c: &Qwen3Config,
+        index: usize,
+        input: &[f32],
+        adapters: &AdapterSet,
+    ) -> Result<(Vec<f32>, Vec<f32>, Vec<f32>)> {
+        let (mut q, mut k, mut v, q_bias, k_bias, v_bias, apply_separate_bias) = match self {
+            Self::Separate {
+                query,
+                key,
+                value,
+                query_bias,
+                key_bias,
+                value_bias,
+            } => (
+                adapters.multiply(&format!("blk.{index}.attn_q.weight"), query, input)?,
+                adapters.multiply(&format!("blk.{index}.attn_k.weight"), key, input)?,
+                adapters.multiply(&format!("blk.{index}.attn_v.weight"), value, input)?,
+                query_bias,
+                key_bias,
+                value_bias,
+                true,
+            ),
+            Self::Fused {
+                projection,
+                bias,
+                query_bias,
+                key_bias,
+                value_bias,
+            } => {
+                let mut qkv = adapters.multiply(
+                    &format!("blk.{index}.attn_qkv.weight"),
+                    projection,
+                    input,
+                )?;
+                math::bias(&mut qkv, bias);
+                let kv = c.kv_width()?;
+                let query_width = c.query_width()?;
+                (
+                    qkv[..query_width].to_vec(),
+                    qkv[query_width..query_width + kv].to_vec(),
+                    qkv[query_width + kv..].to_vec(),
+                    query_bias,
+                    key_bias,
+                    value_bias,
+                    query_bias.is_some() && key_bias.is_some() && value_bias.is_some(),
+                )
+            }
+        };
+        // The source fused graph enables separate Q/K/V biases as one group.
+        // Its separate-projection graph applies each optional bias independently.
+        if apply_separate_bias {
+            math::bias(&mut q, q_bias);
+            math::bias(&mut k, k_bias);
+            math::bias(&mut v, v_bias);
+        }
+        Ok((q, k, v))
+    }
+    fn weight(&self, name: &str) -> Option<&Matrix> {
+        match (self, name) {
+            (Self::Separate { query, .. }, "attn_q.weight") => Some(query),
+            (Self::Separate { key, .. }, "attn_k.weight") => Some(key),
+            (Self::Separate { value, .. }, "attn_v.weight") => Some(value),
+            (Self::Fused { projection, .. }, "attn_qkv.weight") => Some(projection),
+            _ => None,
+        }
+    }
+    fn fingerprint(&self, hash: &mut Fingerprint) {
+        let (biases, matrices) = match self {
+            Self::Separate {
+                query,
+                key,
+                value,
+                query_bias,
+                key_bias,
+                value_bias,
+            } => {
+                hash.bytes(b"separate");
+                (
+                    vec![query_bias, key_bias, value_bias],
+                    vec![query, key, value],
+                )
+            }
+            Self::Fused {
+                projection,
+                bias,
+                query_bias,
+                key_bias,
+                value_bias,
+            } => {
+                hash.bytes(b"fused");
+                (
+                    vec![bias, query_bias, key_bias, value_bias],
+                    vec![projection],
+                )
+            }
+        };
+        for matrix in matrices {
+            hash.matrix(matrix);
+        }
+        for bias in biases {
+            hash.usize(usize::from(bias.is_some()));
+            if let Some(bias) = bias {
+                hash.floats(bias);
+            }
+        }
+    }
+}

+        }
+        let layer = self.weights.layers.get(parsed)?;
+        layer.qkv.weight(field).or(match field {
+            "attn_output.weight" => Some(&layer.attention_output),
+            "ffn_gate.weight" => Some(&layer.gate),
+            "ffn_up.weight" => Some(&layer.up),
+            "ffn_down.weight" => Some(&layer.down),
+            _ => None,
+        })
+    }
+    pub(crate) fn embedding(&self, token: u32) -> Result<Vec<f32>> {
+        if token as usize >= self.config.vocabulary {
+            return Err(Error::InvalidToken {
+                token,
+                vocabulary: self.config.vocabulary,
+            });
+        }
+        Ok(self.weights.token_embedding.row(token as usize)?.to_vec())
+    }
+    pub(crate) fn step(
+        &self,
+        mut hidden: Vec<f32>,
+        position: usize,
+        cache: &mut [KvLayer],
+        adapters: &AdapterSet,
+        cancel: &mut impl FnMut() -> bool,
+    ) -> Result<(Vec<f32>, Vec<f32>)> {
+        let c = &self.config;
+        let head_dim = c.head_dim()?;
+        let scale = (head_dim as f32).sqrt().recip();
+        for (index, (layer, kv)) in self.weights.layers.iter().zip(cache).enumerate() {
+            if cancel() {
+                return Err(Error::Cancelled);
+            }
+            let normalized = math::rms_norm(&hidden, &layer.attention_norm, c.rms_epsilon)?;
+            let (mut q, mut k, v) = layer.qkv.project(c, index, &normalized, adapters)?;
+            normalize_heads(&mut q, &layer.query_norm, head_dim, c.rms_epsilon)?;
+            normalize_heads(&mut k, &layer.key_norm, head_dim, c.rms_epsilon)?;
+            math::rope_neox(&mut q, head_dim, position, c.rope_base);
+            math::rope_neox(&mut k, head_dim, position, c.rope_base);
+            kv.keys.extend(k);
+            kv.values.extend(v);
+            let attention = math::attention(
+                &q, &kv.keys, &kv.values, c.heads, c.kv_heads, head_dim, scale,
+            )?;
+            let projected = adapters.multiply(
+                &format!("blk.{index}.attn_output.weight"),
+                &layer.attention_output,
+                &attention,
+            )?;
+            math::add(&mut hidden, &projected);
+            let normalized = math::rms_norm(&hidden, &layer.feed_forward_norm, c.rms_epsilon)?;
+            let mut gate = adapters.multiply(
+                &format!("blk.{index}.ffn_gate.weight"),
+                &layer.gate,
+                &normalized,
+            )?;
+            let up = adapters.multiply(
+                &format!("blk.{index}.ffn_up.weight"),
+                &layer.up,
+                &normalized,
+            )?;
+            math::native_rows::silu_f32_in_place(&mut gate);
+            for (gate, up) in gate.iter_mut().zip(up) {
+                *gate *= up;
+            }
+            let output =
+                adapters.multiply(&format!("blk.{index}.ffn_down.weight"), &layer.down, &gate)?;
+            math::add(&mut hidden, &output);
+            adapters.control(index, &mut hidden);
+            math::finite(&hidden, "Qwen3 layer output")?;
+        }
+        let embedding = math::rms_norm(&hidden, &self.weights.output_norm, c.rms_epsilon)?;
+        let output = self
+            .weights
+            .output
+            .as_ref()
+            .unwrap_or(&self.weights.token_embedding);
+        let logits = adapters.multiply("output.weight", output, &embedding)?;
+        math::finite(&logits, "Qwen3 logits")?;
+        Ok((embedding, logits))
+    }
+    fn calculate_fingerprint(&self) -> u64 {
+        let mut hash = Fingerprint::new();
+        hash.bytes(b"qwen3-dense-neox-v1");
+        let c = &self.config;
+        for value in [
+            c.vocabulary,
+            c.width,
+            c.feed_forward,
+            c.heads,
+            c.head_dim,
+            c.kv_heads,
+            c.max_context,
+            self.layer_count(),
+        ] {
+            hash.usize(value);
+        }
+        hash.floats(&[c.rms_epsilon, c.rope_base]);
+        hash.matrix(&self.weights.token_embedding);
+        hash.floats(&self.weights.output_norm);
+        hash.usize(usize::from(self.weights.output.is_some()));
+        if let Some(output) = &self.weights.output {
+            hash.matrix(output);
+        }
+        for layer in &self.weights.layers {
+            hash.floats(&layer.attention_norm);
+            layer.qkv.fingerprint(&mut hash);
+            hash.floats(&layer.query_norm);
+            hash.floats(&layer.key_norm);
+            hash.floats(&layer.feed_forward_norm);
+            for matrix in [&layer.attention_output, &layer.gate, &layer.up, &layer.down] {
+                hash.matrix(matrix);
+            }
+        }
+        hash.finish()
+    }
+}

+/// Q/K are shaped [head_dim, heads, token] by the frozen build_qkv graph.
+/// Normalize each contiguous head row with the same learned vector.
+fn normalize_heads(
+    values: &mut [f32],
+    weights: &[f32],
+    head_dim: usize,
+    epsilon: f32,
+) -> Result<()> {
+    if head_dim == 0 || !values.len().is_multiple_of(head_dim) {
+        return Err(Error::Shape("Qwen3 head normalization shape".into()));
+    }
+    for head in values.chunks_mut(head_dim) {
+        let normalized = math::rms_norm(head, weights, epsilon)?;
+        head.copy_from_slice(&normalized);
+    }
+    Ok(())

+// Frozen GGML binary operations keep the F32 activation as src0. Their CPU
+// mixed-type branches cannot consume a low-precision active norm/bias as src1.
+// Validate these descriptors before decoding any weights. Partial fused bias
+// groups are deliberately ignored by the original graph, so their stored types
+// remain admissible and are decoded only for checked model ownership/identity.
+fn require_f32_if_present(g: &GgufSet, name: &str) -> Result<()> {
+    if let Some(tensor) = g.tensor(name)
+        && tensor.dtype.id() != 0
+    {
+        return Err(unsupported(format!(
+            "Qwen3 active norm/bias tensor {name} requires F32 storage"
+        )));
+    }
+    Ok(())
+}

+fn validate_active_vector_storage(g: &GgufSet, layers: usize) -> Result<()> {
+    require_f32_if_present(g, "output_norm.weight")?;
+    for index in 0..layers {
+        let key = |name: &str| format!("blk.{index}.{name}");
+        for name in [
+            "attn_norm.weight",
+            "attn_q_norm.weight",
+            "attn_k_norm.weight",
+            "ffn_norm.weight",
+        ] {
+            require_f32_if_present(g, &key(name))?;
+        }
+        let biases = [key("attn_q.bias"), key("attn_k.bias"), key("attn_v.bias")];
+        if g.tensor(&key("attn_qkv.weight")).is_none() {
+            for name in &biases {
+                require_f32_if_present(g, name)?;
+            }
+        } else if g.tensor(&key("attn_qkv.bias")).is_some() {
+            require_f32_if_present(g, &key("attn_qkv.bias"))?;
+        } else if biases.iter().all(|name| g.tensor(name).is_some()) {
+            for name in &biases {
+                require_f32_if_present(g, name)?;
+            }
+        }
+    }
+    Ok(())

+pub(super) fn load(mut g: GgufSet, path: &Path, options: &LoadOptions) -> Result<Model> {
+    let metadata = g.metadata();
+    match metadata.get("general.type") {
+        None => {}
+        Some(Value::String(kind)) if kind == "model" => {}
+        Some(Value::String(kind)) => return Err(unsupported(format!("GGUF general.type={kind}"))),
+        Some(_) => return Err(Error::Invalid("general.type must be a string".into())),
+    }
+    let tokenizer = Tokenizer::from_metadata(metadata)?;
+    let layers = uint(metadata, "qwen3.block_count", None)?;
+    if layers == 0 || layers > 512 {
+        return Err(Error::Invalid("layer count must be 1..=512".into()));
+    }
+    let train_context = uint(metadata, "qwen3.context_length", None)?;
+    if !metadata.contains_key("qwen3.attention.layer_norm_rms_epsilon") {
+        return Err(Error::Invalid(
+            "missing qwen3.attention.layer_norm_rms_epsilon".into(),
+        ));
+    }
+    for key in [
+        "qwen3.attention.head_count",
+        "qwen3.attention.head_count_kv",
+        "qwen3.feed_forward_length",
+    ] {
+        if matches!(metadata.get(key), Some(Value::Array { .. })) {
+            return Err(unsupported(format!("per-layer array {key}")));
+        }
+    }
+    let heads = uint(metadata, "qwen3.attention.head_count", None)?;
+    let width = uint(metadata, "qwen3.embedding_length", None)?;
+    let default_head = width
+        .checked_div(heads)
+        .ok_or_else(|| Error::Invalid("qwen3 attention head count must be positive".into()))?;
+    let key_length = uint(metadata, "qwen3.attention.key_length", Some(default_head))?;
+    let value_length = uint(metadata, "qwen3.attention.value_length", Some(default_head))?;
+    let config = Qwen3Config {
+        vocabulary: tokenizer.vocabulary_size(),
+        width,
+        feed_forward: uint(metadata, "qwen3.feed_forward_length", None)?,
+        heads,
+        head_dim: key_length,
+        kv_heads: uint(metadata, "qwen3.attention.head_count_kv", Some(heads))?,
+        max_context: train_context.min(options.context_length),
+        rms_epsilon: float(metadata, "qwen3.attention.layer_norm_rms_epsilon", 1e-5)?,
+        rope_base: float(metadata, "qwen3.rope.freq_base", 10000.)?,
+    };
+    config.validate()?;
+    let head_dim = config.head_dim()?;
+    let kv_width = config.kv_width()?;
+    let query_width = config.query_width()?;
+    if value_length != head_dim {
+        return Err(unsupported(
+            "Qwen3 requires equal key and value head lengths",
+        ));
+    }
+    if uint(metadata, "qwen3.rope.dimension_count", Some(head_dim))? != head_dim {
+        return Err(unsupported("Qwen3 requires full-head NeoX RoPE"));
+    }
+    for key in [
+        "qwen3.attention.sliding_window",
+        "qwen3.expert_count",
+        "qwen3.expert_used_count",
+    ] {
+        if uint(metadata, key, Some(0))? != 0 {
+            return Err(unsupported(format!("nonzero {key}")));
+        }
+    }
+    require_causal(metadata, "qwen3", true)?;
+    require_unscaled_rope(metadata, "qwen3")?;
+    if uint(metadata, "qwen3.pooling_type", Some(0))? != 0 {
+        return Err(unsupported("Qwen3 pooled embedding or reranker execution"));
+    }
+    for tensor in g.tensors() {
+        if tensor.name.starts_with("cls.")
+            || tensor.name.starts_with("dense_2.")
+            || tensor.name == "output.bias"
+        {

+fn compare_row(row: &[f32], data: &[u8], offset: usize, label: &str, metrics: &mut Metrics) {
+    let raw = &data[offset..offset.checked_add(row.len() * 4).unwrap()];
+    for (index, (&actual, expected)) in row.iter().zip(raw.as_chunks::<4>().0).enumerate() {
+        let expected = f32::from_le_bytes(*expected);
+        let difference = (f64::from(actual) - f64::from(expected)).abs();
+        let budget = 2e-4 + 2e-5 * f64::from(expected).abs();
+        assert!(
+            actual.is_finite() && expected.is_finite() && difference <= budget,
+            "{label} col={index}: actual={actual:.9e} expected={expected:.9e} difference={difference:.9e} budget={budget:.9e}"
+        );
+        if cfg!(all(
+            target_arch = "x86_64",
+            any(
+                all(target_os = "windows", target_env = "msvc"),
+                all(target_os = "linux", target_env = "gnu")
+            )
+        )) {
+            assert_eq!(
+                actual.to_bits(),
+                expected.to_bits(),
+                "{label} col={index}: measured native profile must match exactly"
+            );
+        }
+        metrics.floats += 1;
+        metrics.bitwise += usize::from(actual.to_bits() == expected.to_bits());
+        metrics.maximum = metrics.maximum.max(difference);
+        metrics.fraction = metrics.fraction.max(difference / budget);
+    }
+    metrics.rows += 1;

+# Dense causal Qwen3 implementation
+
+This is a bounded CPU implementation of the dense Qwen3 language-model graph in the frozen llama.cpp source archive. It adds Qwen3 to the Rust inference facade and strict transactional Context. It does not close the full architecture, backend, application or full-port requirements.
+
+## Source and model identity
+
+The source oracle is the user's frozen archive, SHA-256 `2db1e23f619fcabfd628481683c57b1e3e90db86bd045900a04b6073e1843122`. The graph follows `src/models/qwen3.cpp`, the shared `create_tensor_qkv` loading rules, and the original CPU operators. The existing upstream license and notice apply to this translation.
+
+The pretrained reference is the official [Qwen/Qwen3-0.6B-GGUF](https://huggingface.co/Qwen/Qwen3-0.6B-GGUF/tree/23749fefcc72300e3a2ad315e1317431b06b590a) file `Qwen3-0.6B-Q8_0.gguf`, pinned to revision `23749fefcc72300e3a2ad315e1317431b06b590a`. Its 639,446,688 bytes have SHA-256 `9465e63a22add5354d9bb4b99e90117043c7124007664907259bd16d043bb031`. The exact Apache-2.0 license from that revision is preserved in `fixtures/inference/qwen3/LICENSE.Qwen3.txt`. Downloaded pretrained weights remain external to the repository.
+
+## Graph contracts
+
+`Qwen3Config` represents head width independently from embedding width. Query width is `heads * head_dim`; key/value width is `kv_heads * head_dim`. The model therefore supports the official configuration with embedding width 1,024, 16 query heads and head dimension 128: its query projection width is 2,048. Shapes, grouping, integer products and finite configuration values are checked before execution.
+
+Each query and key head is normalized with its learned RMS vector before full-head NeoX RoPE. The attention output projection maps the query width back to embedding width. The graph retains the frozen RMS, matrix, attention and SiLU arithmetic profiles; it does not substitute Qwen2's configuration or omit Q/K normalization.
+
+Separate Q/K/V projections apply each present bias independently. A fused QKV projection uses its own bias when present. When the fused bias is absent, the original graph applies separate Q/K/V biases only if all three exist. Partial bias groups in that fused case remain inactive. Tied and untied output projections are supported.
+
+Active normalization vectors and biases require F32 storage in this CPU graph. The original GGML binary operation keeps the F32 activation as its first operand and does not support a low-precision second operand for these operations. The loader checks active descriptors before weight decoding. Inactive partial fused bias groups retain their original acceptance behavior; the loader still validates their shapes and ownership.
+
+The loader uses the validated `GgufSet` owner, primary-shard metadata and per-shard tensor offsets. Auto-discovered and explicit split lists are covered, including a first shard containing metadata and no tensors. No metadata from later shards is merged into the primary metadata.
+
+## State and resource behavior
+
+The strict Context uses the actual Qwen3 KV width for growth, truncation, sequence copying and serialized state. Its model fingerprint includes the Qwen3 graph tag, head dimension, Q/K normalization vectors, biases and matrix storage identities. Checkpoints and restored states cannot silently cross different Qwen3 model identities.
+
+Cancellation, allocation failures and nonfinite execution leave the committed batch state unchanged. The existing explicit default 1 GiB state payload/copy budget remains a per-context ledger, not a total-process memory cap. It excludes execution scratch, output buffers, allocator overhead and caller-owned serialized input.
+
+The inference facade's default decoded-weight budget remains 2 GiB. The pinned pretrained model needs 2,384,199,680 decoded-weight bytes, so its test explicitly requests a 3 GiB budget. A normal default-budget load is required to reject it before weight decoding. Requested test context sizes are 64 for synthetic models and 128 for the pretrained model; the original runtime may pad its physical allocation.
+
+## Independent regressions
+
+Both expected profiles were captured through the original C/C++ library before Rust comparisons. They are selected by the independently compiled target, not by a runtime option. The compact fixtures retain exact input GGUF bytes, input hashes, original token IDs and pieces, exit status, and round-trip F32 logits. Identical rows and identical input blobs are interned only after exact byte comparison. No expected numeric value is generated by the Rust implementation.
+
+| Corpus | Cases and checks |
+| --- | --- |
+| Dense graph and splits | 51 configurations: 39 admitted, 8 explicitly unsupported, 3 invalid, and 1 checked nonfinite execution. Admitted cases compare 279,552 logits and 390 greedy IDs per profile. |
+| Active vector storage | 17 configurations: 12 native rejection cases, an F32 control, and 4 valid ignored partial fused-bias cases. The valid cases compare 15,360 logits and 10 greedy IDs per profile. |
+| Official pretrained Q8_0 | Three prompts, 5,469,696 nonduplicated logits and 24 greedy IDs per profile. The complete tokenizer pieces, generated bytes and generation statistics are also checked. |
+
+The tiny graph corpus covers independent attention widths, MHA/MQA/GQA, an embedding width not divisible by head count, learned Q/K norms, all separate and fused partial-bias subsets, fused own bias, tied/untied outputs, F32/F16/Q4_0/Q4_K projection cases, and implicit key/value length defaults. Q8_0 projection execution is exercised by the pretrained model.
+
+The original Windows MSVC and Linux GNU profiles have separate expected data. The measured x86_64 targets enforce exact F32 bits, token IDs and piece bytes. Other targets retain the predeclared portable bound `2e-4 + 2e-5 * abs(reference)`; this is not evidence that unmeasured targets have been validated.

+The storage rejection regression preserves the initial Rust failure: it previously admitted all 12 active low-precision vector cases that the original CPU graph rejected. +## Explicit remaining scope
+
+Per-layer dimension arrays, unequal key/value head lengths, partial or scaled RoPE, nonunit attention gain, noncausal attention, sliding-window attention, MoE, classifier/reranker and additional embedding projections are outside this slice and return explicit unsupported errors when encountered. Qwen3 MoE, VL, Next and other separate architectures remain separate port work.
+
+The broader C NativeContext execution path remains Llama-only; adding a Rust inference architecture does not imply C execution parity. GPU execution and the original scheduler's full batching behavior remain separate work. Existing chat-template handling does not establish full Qwen3 thinking, tool-use or Jinja parity.
+
+The new storage preflight is specific to the Qwen3 loader. It does not claim that every other architecture's active vector storage acceptance has been audited.
